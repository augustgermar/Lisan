385def embedding_settings(config: dict[str, Any]) -> dict[str, Any]: 

386 """Resolve the retrieval.embeddings block, backfilling any missing keys 

387 from the defaults so older config files keep working.""" 

388 defaults = DEFAULT_CONFIG["retrieval"]["embeddings"] 

389 merged = deepcopy(defaults) 

390 block = config.get("retrieval", {}).get("embeddings", {}) 

391 if isinstance(block, dict): 

392 merged.update(block) 

393 return merged

396def load_config(path: Path | None = None) -> dict[str, Any]: 

397 cfg_path = path or config_path() 

398 if not cfg_path.exists(): 

399 return deepcopy(DEFAULT_CONFIG) 

400 raw = cfg_path.read_text(encoding="utf-8").strip() 

401 if not raw: 

402 return deepcopy(DEFAULT_CONFIG) 

403 data = json.loads(raw) 

404 merged = deepcopy(DEFAULT_CONFIG) 

405 _deep_merge(merged, data) 

406 _normalize_local_provider_defaults(merged) 

407 return merged

410def save_default_config(path: Path | None = None) -> Path: 

411 cfg_path = path or config_path() 

412 cfg_path.parent.mkdir(parents=True, exist_ok=True) 

413 cfg_path.write_text(json.dumps(DEFAULT_CONFIG, indent=2), encoding="utf-8") 

414 return cfg_path

417def _deep_merge(base: dict[str, Any], overlay: dict[str, Any]) -> None: 

418 for key, value in overlay.items(): 

419 if isinstance(value, dict) and isinstance(base.get(key), dict): 

420 _deep_merge(base[key], value) 

421 else: 

422 base[key] = value
