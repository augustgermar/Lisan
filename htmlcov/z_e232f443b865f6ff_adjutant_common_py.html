21# Schedule cadence. The first two forms are the existing scheduler's 

22# vocabulary (scheduler.normalize_recurrence); the weekly/monthly forms 

23# are the Adjutant's additions, taught to next_occurrence when schedule

25_CRON_RES = [ 

26 re.compile(r"^every:\d+[smhdw]$"), 

27 re.compile(r"^daily@([01]?\d|2[0-3]):[0-5]\d$"),

45def next_cron_occurrence(cron: str, *, after=None): 

46 """First fire time strictly after ``after`` (default now), as an aware 

47 UTC datetime. Handles all four cadence forms; every:/daily@ delegate to 

48 the existing scheduler (one home per form). The cadence is always 

49 computed from the record's cron string — it never migrates into the DB.""" 

50 from datetime import timedelta 

51 

52 from .scheduler import _local_tz, _now_utc, next_occurrence 

53 

54 text = str(cron).strip().lower() 

55 if text.startswith("every:") or text.startswith("daily@"): 

56 return next_occurrence(text, after=after) 

57 after = after or _now_utc() 

58 local_after = after.astimezone(_local_tz()) 

59 

60 match = _WEEKLY_RE.match(text) 

61 if match: 

62 target_weekday = _WEEKDAYS.index(match.group(1)) 

63 hour, minute = int(match.group(2)), int(match.group(3)) 

64 candidate = local_after.replace(hour=hour, minute=minute, second=0, microsecond=0) 

65 candidate += timedelta(days=(target_weekday - local_after.weekday()) % 7) 

66 if candidate <= local_after: 

67 candidate += timedelta(days=7) 

68 return candidate.astimezone(_utc()) 

69 

70 match = _MONTHLY_RE.match(text) 

71 if match: 

72 day, hour, minute = int(match.group(1)), int(match.group(2)), int(match.group(3)) 

73 candidate = local_after.replace(day=day, hour=hour, minute=minute, second=0, microsecond=0) 

74 if candidate <= local_after: 

75 month = local_after.month + 1 

76 year = local_after.year + (month - 1) // 12 

77 month = (month - 1) % 12 + 1 

78 candidate = candidate.replace(year=year, month=month) 

79 return candidate.astimezone(_utc()) 

80 

81 raise ValueError(f"unsupported cadence {cron!r}")
