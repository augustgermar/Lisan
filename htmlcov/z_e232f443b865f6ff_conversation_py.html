1"""The conversational agent: one tool-bearing call per turn, full history. 

2 

3This is the conversation layer's spine. Every non-trivial turn — question, 

4story, request, aside — goes to a single agent that sees the rolling 

5conversation verbatim, retrieved memory context, and its own capability 

6index, with every tool available. It answers in one model call; the memory 

7pipeline runs afterwards, in the background, as an observer of the finished 

8exchange (the `capture.observe` job). 

9 

10The design lesson behind it: a router in front of the model misroutes, and 

11an agent that never sees the conversation cannot hold a thread. The model 

12routes implicitly, with context; memory capture never again stands between 

13the user and the reply. 

14"""

37def run_conversation_turn( 

38 *, 

39 vault: Path | None = None, 

40 text: str, 

41 conversation_id: str | None = None, 

42 provider: str | None = None, 

43 model: str | None = None, 

44 db_path: Path | None = None, 

45 approval_fn: Callable[[str, dict[str, Any]], bool] | None = None, 

46 queue_capture: bool = True, 

47 capture_text: str | None = None, 

48) -> dict[str, Any]: 

49 """One conversational turn: transcript in, one agent call, transcript out, 

50 capture observed in the background. 

51 

52 ``queue_capture=False`` is the /forget contract: the exchange is answered 

53 but never handed to the memory pipeline. ``capture_text`` lets the caller 

54 give the pipeline the owner's original words (e.g. with the /remember 

55 flag the heuristics key on) while the agent sees the stripped text.""" 

56 from ..agents.conversation import ConversationAgent 

57 

58 vault = vault or vault_root() 

59 record_inline_step("conversation.turn") 

60 append_transcript(vault=vault, conversation_id=conversation_id, speaker="USER", text=text) 

61 

62 # Fail visibly if the transcript write succeeded but the conversation 

63 # reader cannot reconstruct the turn. This protects the live path from 

64 # silently degrading back into stateless behavior after a parser or vault 

65 # regression. 

66 persisted: list[dict[str, str]] | None = None 

67 try: 

68 persisted = conversation_history(vault, conversation_id) 

69 if not any(turn.get("speaker") == "USER" and turn.get("text") == text for turn in persisted): 

70 log_error( 

71 vault, 

72 "conversation.transcript_round_trip", 

73 ValueError("new user turn was written but could not be read back"), 

74 ) 

75 except Exception as exc: 

76 log_error(vault, "conversation.transcript_round_trip", exc) 

77 

78 history = _rolling_history(vault, conversation_id) 

79 context = _retrieval_context(vault=vault, text=text, conversation_id=conversation_id, db_path=db_path) 

80 profile = _owner_profile(vault) 

81 self_story = _self_story_context(vault, text) 

82 if self_story: 

83 context = f"{context}\n\n{self_story}" if context else self_story 

84 active_focus = _active_ingestion_context(vault) 

85 if active_focus: 

86 context = f"{context}\n\n{active_focus}" if context else active_focus 

87 

88 # Session open is the drive system's single delivery seam (v1 action 

89 # budget): a fresh conversation may carry at most one question-phrased 

90 # callback. The user's turn is already in the transcript by now, so 

91 # "open" means this conversation holds exactly that one turn. The drive 

92 # must never break a conversation turn. 

93 unresolved_thread = None 

94 try: 

95 # `persisted` was read just above, after this turn was appended. 

96 if persisted is not None and len(persisted) <= 1: 

97 from .drive import session_open_callback 

98 

99 unresolved_thread = session_open_callback(vault, conversation_id) 

100 except Exception as exc: 

101 log_error(vault, "conversation.drive_callback", exc) 

102 

103 # WO-GROUND Seam A: on a self-referential turn the model does not get to 

104 # choose whether to consult ground truth — the truth is already in front 

105 # of it before it answers. (The self_state tool stays, for turns the 

106 # deterministic detector misses.) 

107 ground_truth = None 

108 try: 

109 from .self_questions import detect_self_question, render_ground_truth 

110 

111 needed = detect_self_question(text) 

112 if needed: 

113 ground_truth = render_ground_truth(needed, vault=vault, db_path=db_path) 

114 except Exception as exc: 

115 log_error(vault, "conversation.ground_truth", exc) 

116 

117 # IIP Phase 1: an interpretation-of-a-person turn carries a hard protocol 

118 # — locus-diverse hypotheses, discriminators, a convergent action — 

119 # injected as a directive and enforced deterministically after the call. 

120 # Independent of GROUND_TRUTH: a turn about a person AND the system 

121 # ("why does she keep asking about my reminders?") carries both blocks. 

122 interpretation_directive = None 

123 try: 

124 from .interpretation import INTERPRETATION_DIRECTIVE, is_interpretation_query 

125 

126 if is_interpretation_query(text): 

127 interpretation_directive = INTERPRETATION_DIRECTIVE 

128 except Exception as exc: 

129 log_error(vault, "conversation.interpretation_detect", exc) 

130 

131 agent = ConversationAgent(vault=vault) 

132 record_inline_step("conversation.agent") 

133 

134 # Section order is cache order (render_input preserves kwargs order): 

135 # stable blocks first (capabilities, owner profile), volatile last 

136 # (retrieval, the growing conversation, and the minute-resolution 

137 # clock) — so a provider's prefix cache survives across turns instead 

138 # of missing on the first volatile byte. validator_feedback, when a 

139 # regeneration needs it, appends after everything. 

140 def _call_agent(validator_feedback: str | None = None) -> dict[str, Any]: 

141 return agent.run_json( 

142 json.dumps({"user_message": text}, indent=2, ensure_ascii=True), 

143 significance="medium", 

144 provider=provider, 

145 model=model, 

146 provider_error_mode="raise", 

147 capabilities=cached_capability_index(), 

148 owner_profile=profile or None, 

149 retrieved_context=context or None, 

150 ground_truth=ground_truth, 

151 interpretation_protocol=interpretation_directive, 

152 unresolved_thread=unresolved_thread, 

153 conversation=history or None, 

154 today=_today_line(), 

155 validator_feedback=validator_feedback, 

156 db_path=db_path, 

157 conversation_id=conversation_id, 

158 approval_fn=approval_fn, 

159 ) 

160 

161 out = _call_agent() 

162 if interpretation_directive is not None: 

163 out = _enforce_interpretation_protocol( 

164 out, _call_agent, text=text, vault=vault, db_path=db_path, 

165 ) 

166 response = str(out.get("response") or "").strip() 

167 tool_calls = list(getattr(agent, "last_tool_calls", []) or []) 

168 if not response: 

169 log_error(vault, "conversation.empty_response", ValueError(f"empty response for: {text[:120]!r}")) 

170 retry_out = _call_agent() 

171 response = str(retry_out.get("response") or "").strip() 

172 tool_calls = list(getattr(agent, "last_tool_calls", []) or []) 

173 if not response: 

174 log_error(vault, "conversation.empty_response_retry", ValueError(f"retry also empty for: {text[:120]!r}")) 

175 response = ( 

176 "I wasn't able to process that — can you try again?" 

177 ) 

178 

179 append_transcript(vault=vault, conversation_id=conversation_id, speaker="LISAN", text=response) 

180 

181 queued: list[dict[str, Any]] = [] 

182 if queue_capture: 

183 queued_job = _queue_observation( 

184 vault=vault, 

185 text=capture_text or text, 

186 response=response, 

187 tool_calls=tool_calls, 

188 conversation_id=conversation_id, 

189 db_path=db_path, 

190 ) 

191 if queued_job: 

192 queued.append(queued_job) 

193 record_jobs_queued(1) 

194 

195 return { 

196 "response": response, 

197 "route": "conversation", 

198 "tool_calls": tool_calls, 

199 "queued_jobs": queued, 

200 }

211 """Who the user is, always in context: the primer profile plus the 

212 identity-core roster. Kinship shorthand ("the girls", "my brother") can 

213 only ground against an always-present cast — retrieval echoes are too

217 identity_path = vault / "primer" / "identity.md" 

218 identity = identity_path.read_text(encoding="utf-8").strip() 

219 if identity: 

220 body = identity.split("---")[-1].strip() 

221 parts.append("PRINCIPAL_PROFILE\n" + file_line(identity_path, default_ttl_days=30) + "\n" + body[:1500]) 

222 except Exception as exc: 

223 log_error(vault, "conversation.owner_profile identity load failed", exc) 

224 try: 

225 from .primer_index import _identity_core 

226 

227 core = _identity_core(vault) or {} 

228 roster = core.get("roster") or [] 

229 if roster: 

230 people = "; ".join( 

231 f"{r.get('name')} ({r.get('relation')})" if isinstance(r, dict) else str(r) 

232 for r in roster 

233 ) 

234 parts.append("IDENTITY_CORE\n" + file_line(vault / "primer" / "identity-core.md", stable=True) + f"\nHousehold cast: {people}") 

235 except Exception as exc: 

236 log_error(vault, "conversation.owner_profile roster load failed", exc) 

237 try: 

238 style = (vault / "primer" / "operating-style.md").read_text(encoding="utf-8")

242 if standing: 

243 parts.append("Standing instructions from the user about how to behave " 

244 "(honor these every turn):\n" + file_line(vault / "primer" / "operating-style.md", default_ttl_days=30) + "\n" + standing[:1200]) 

245 except Exception as exc: 

246 log_error(vault, "conversation.owner_profile operating-style load failed", exc) 

247 parts.append(_self_identity_line(vault)) 

248 return "\n\n".join(p for p in parts if p)

259def _self_story_context(vault: Path, text: str) -> str: 

260 """When the user asks about the agent itself, inject the autobiography: 

261 recent self-episodes (Layer B), ratified beliefs, and voice provenance. 

262 Deterministic — assembled from records, so the story cannot be 

263 confabulated; a thin Layer B yields a thin (honest) story.""" 

264 if not _SELF_STORY_TRIGGER.search(text or ""): 

265 return "" 

266 parts: list[str] = [] 

267 episodes_dir = vault / "self" / "episodes" 

268 if episodes_dir.exists(): 

269 from ..frontmatter import load_markdown 

270 

271 rows = [] 

272 for p in sorted(episodes_dir.glob("*.md"), reverse=True)[:12]: 

273 try: 

274 fm = load_markdown(p).frontmatter 

275 rows.append(f"- ({fm.get('created')}) {fm.get('summary')}") 

276 except Exception: 

277 continue 

278 if rows: 

279 parts.append("Recent events in your own life (deterministic records — safe to narrate):\n" + "\n".join(rows)) 

280 beliefs_dir = vault / "self" / "beliefs" 

281 if beliefs_dir.exists(): 

282 from ..frontmatter import load_markdown 

283 

284 rows = [] 

285 for p in sorted(beliefs_dir.glob("*.md"))[:8]: 

286 try: 

287 fm = load_markdown(p).frontmatter 

288 rows.append(f"- {fm.get('summary') or p.stem}") 

289 except Exception: 

290 continue 

291 if rows: 

292 parts.append("Your ratified self-beliefs:\n" + "\n".join(rows)) 

293 try: 

294 core = (vault / "primer" / "identity-core.md").read_text(encoding="utf-8") 

295 if "Voice Provenance" in core: 

296 prov = core.split("## Voice Provenance", 1)[1].strip()[:500] 

297 parts.append("How your voice was formed (provenance):\n" + prov) 

298 except Exception: 

299 pass 

300 if not parts: 

301 return "" 

302 return "SELF_STORY (your own autobiography, from your records):\n\n" + "\n\n".join(parts)

305def _self_identity_line(vault: Path) -> str: 

306 """The assistant's name identity, stated deterministically from the 

307 kernel every turn. The prompt's {{self}} token renders as the nickname, 

308 while kernel materials carry the canonical name — without this line the 

309 model reconciles the two by disowning one of its own names (observed at 

310 the first live Wipe Test: the agent disowned its own canonical name).""" 

311 try: 

312 from .primer_index import _identity_core 

313 

314 assistant = (_identity_core(vault) or {}).get("assistant") or {} 

315 canonical = str(assistant.get("canonical_name") or assistant.get("name") or "").strip() 

316 nickname = str(assistant.get("nickname") or "").strip() 

317 if not canonical: 

318 return "" 

319 freshness = file_line(vault / "primer" / "identity-core.md", stable=True) 

320 if nickname and nickname != canonical: 

321 return (f"Your identity kernel: your canonical name is {canonical}; you go by {nickname}. " 

322 f"{freshness} Both are your names.") 

323 return f"Your identity kernel: your name is {canonical}. {freshness}" 

324 except Exception as exc: 

325 log_error(vault, "conversation.self_identity load failed", exc) 

326 return ""

329def _rolling_history(vault: Path, conversation_id: str | None) -> str: 

330 """The last stretch of this conversation, verbatim. The agent must see 

331 the actual back-and-forth — summaries alone cannot hold a thread.""" 

332 try: 

333 turns = conversation_history(vault, conversation_id) 

334 except Exception: 

335 return "" 

336 if not turns: 

337 return "" 

338 turns = turns[-_HISTORY_TURNS:] 

339 lines = [] 

340 for turn in turns: 

341 speaker = str(turn.get("speaker") or "").strip() or "USER" 

342 lines.append(f"{speaker}: {turn.get('text', '')}") 

343 history = "\n".join(lines) 

344 if len(history) > _HISTORY_CHARS: 

345 history = history[-_HISTORY_CHARS:] 

346 cut = history.find("\n") 

347 if cut > 0: 

348 history = history[cut + 1:] 

349 return history

374def _active_ingestion_context(vault: Path) -> str: 

375 """If files or documents were recently assimilated into the vault (last 24h), 

376 keep a concise focus block in context so follow-up questions have 

377 immediate working awareness of those documents without relying solely 

378 on cold vector retrieval.""" 

379 try: 

380 path = vault / "state" / "active_ingestion.json" 

381 if not path.exists(): 

382 return "" 

383 data = json.loads(path.read_text(encoding="utf-8")) 

384 if not isinstance(data, dict): 

385 return "" 

386 timestamp = data.get("timestamp") 

387 if timestamp: 

388 from datetime import datetime 

389 try: 

390 ingested_at = datetime.fromisoformat(timestamp) 

391 if (datetime.now() - ingested_at).total_seconds() > 86400: 

392 return "" 

393 except Exception: 

394 pass 

395 source = data.get("source_path") or "recent import" 

396 summary = data.get("summary") or "" 

397 files = data.get("files") or [] 

398 lines = [f"RECENT_INGESTION_FOCUS (documents recently assimilated from {source}):"] 

399 if summary: 

400 lines.append(f"Summary: {summary}") 

401 if files: 

402 file_sample = files[:12] 

403 lines.append("Key files:") 

404 for f in file_sample: 

405 lines.append(f"- {f}") 

406 if len(files) > 12: 

407 lines.append(f"- ... and {len(files) - 12} more files") 

408 return "\n".join(lines) 

409 except Exception: 

410 return ""

413def _enforce_interpretation_protocol( 

414 out: dict[str, Any], 

415 call_agent, 

416 *, 

417 text: str, 

418 vault: Path, 

419 db_path: Path | None, 

420) -> dict[str, Any]: 

421 """Deterministic IIP enforcement (never an LLM judging an LLM): validate 

422 the structured hypothesis space, regenerate with the validator's 

423 complaint at most `iip.max_regenerations` times (owner-set: 1), and on 

424 exhaustion render the best attempt with an explicit incompleteness 

425 notice — the requirement is never silently dropped. Every detector fire 

426 is logged, pass or fail, so detector precision stays visible. The 

427 whole ladder is disabled at runtime by `iip.validator_enabled`.""" 

428 from ..config import load_config 

429 from .interpretation import ( 

430 incompleteness_notice, 

431 log_iip_event, 

432 query_digest, 

433 validate_interpretation, 

434 ) 

435 

436 cfg = load_config().get("iip") or {} 

437 event: dict[str, Any] = {"detector": "interpretation", "query_digest": query_digest(text)} 

438 try: 

439 if not cfg.get("validator_enabled", True): 

440 event["validated"] = "disabled" 

441 return out 

442 max_regens = int(cfg.get("max_regenerations", 1)) 

443 complaints = validate_interpretation(out, db_path=db_path) 

444 event["first_pass_complaints"] = list(complaints) 

445 regens = 0 

446 while complaints and regens < max_regens: 

447 regens += 1 

448 feedback = ( 

449 "Your previous answer failed the interpretation protocol: " 

450 + "; ".join(complaints) 

451 + ". Produce the full response again with a compliant interpretation object. " 

452 "Do not mention this correction to the user." 

453 ) 

454 try: 

455 out = call_agent(feedback) 

456 except Exception as exc: 

457 log_error(vault, "conversation.iip_regeneration", exc) 

458 break 

459 complaints = validate_interpretation(out, db_path=db_path) 

460 

461 # IIP Phase 2: a structurally valid answer can still be wholly 

462 # in-register — every hypothesis wearing one of the corpus's own 

463 # favorite frames. The challenge shares the regeneration budget. 

464 if not complaints: 

465 try: 

466 from .attribution_priors import ( 

467 challenge_feedback, 

468 hypotheses_all_in_register, 

469 load_attribution_register, 

470 ) 

471 

472 register = load_attribution_register(vault) 

473 matched = hypotheses_all_in_register(out, register) 

474 if matched: 

475 challenge: dict[str, Any] = {"priors_matched": matched} 

476 event["challenge"] = challenge 

477 if regens < max_regens: 

478 regens += 1 

479 try: 

480 out = call_agent(challenge_feedback(matched)) 

481 complaints = validate_interpretation(out, db_path=db_path) 

482 still = None if complaints else hypotheses_all_in_register(out, register) 

483 challenge["outcome"] = ( 

484 "structure_regressed" if complaints 

485 else ("still_in_register" if still else "cleared") 

486 ) 

487 except Exception as exc: 

488 log_error(vault, "conversation.iip_challenge_regen", exc) 

489 challenge["outcome"] = "regeneration_failed" 

490 else: 

491 challenge["outcome"] = "exhausted" 

492 if challenge["outcome"] in {"exhausted", "still_in_register"}: 

493 response = str(out.get("response") or "").strip() 

494 if response: 

495 out = dict(out) 

496 out["response"] = ( 

497 f"{response}\n\n(A note on my own reasoning: every reading " 

498 "here leans on frames my records already favor about these " 

499 "people — hold them extra loosely.)" 

500 ) 

501 except Exception as exc: 

502 log_error(vault, "conversation.iip_challenge", exc) 

503 

504 event["regenerations"] = regens 

505 if complaints: 

506 event["validated"] = "incomplete" 

507 event["final_complaints"] = list(complaints) 

508 response = str(out.get("response") or "").strip() 

509 if response: 

510 out = dict(out) 

511 out["response"] = f"{response}\n\n{incompleteness_notice(complaints)}" 

512 else: 

513 event["validated"] = "pass" 

514 return out 

515 finally: 

516 log_iip_event(vault, event)

519def _queue_observation( 

520 *, 

521 vault: Path, 

522 text: str, 

523 response: str, 

524 tool_calls: list[dict[str, Any]], 

525 conversation_id: str | None, 

526 db_path: Path | None, 

527) -> dict[str, Any] | None: 

528 """Memory capture as an observer: the exchange is finished; the pipeline 

529 extracts what to remember without the user waiting on it.""" 

530 try: 

531 from .db import retry_locked 

532 from .jobs import enqueue_job 

533 

534 payload = { 

535 "vault": str(vault), 

536 "text": text, 

537 "response": response, 

538 "tool_calls": _compact_tool_calls(tool_calls), 

539 # For the learning loop only: the pipeline above reads the compact 

540 # form (it goes into a prompt), but what an executor reported back is 

541 # exactly what a reviewer needs and 1 in 5 real results was cut at 

542 # 1500 characters. Generous caps; never read by the memory pipeline. 

543 "tool_calls_full": _full_tool_calls(tool_calls), 

544 "conversation_id": conversation_id, 

545 } 

546 # Retried, not just best-effort: dropping this silently is a real 

547 # memory-capture loss (the turn is never seen again), not inert 

548 # telemetry — worth a few hundred ms of retry against transient 

549 # writer contention (e.g. a reindex mid-chunk). 

550 job_id = retry_locked(lambda: enqueue_job("capture.observe", payload, db_path=db_path)) 

551 return {"job_type": "capture.observe", "job_id": job_id} 

552 except Exception as exc: 

553 log_error(vault, "conversation.queue_observation", exc) 

554 return None

561def _full_tool_calls(tool_calls: list[dict[str, Any]]) -> list[dict[str, Any]]: 

562 out = [] 

563 for call in tool_calls[:_FULL_CALLS]: 

564 result = str(call.get("result") or "") 

565 if len(result) > _FULL_RESULT_CHARS: 

566 result = result[: _FULL_RESULT_CHARS - 1] + "…" 

567 out.append({"tool": call.get("tool"), "args": call.get("args"), "result": result}) 

568 return out

571def _compact_tool_calls(tool_calls: list[dict[str, Any]]) -> list[dict[str, Any]]: 

572 compact = [] 

573 for call in tool_calls[:10]: 

574 result = str(call.get("result") or "") 

575 compact.append({
