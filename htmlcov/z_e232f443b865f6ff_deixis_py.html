8Audience = Literal["interlocutor", "substrate", "display"] 

9# interlocutor -> conscious surface: {{principal}}->"you", {{self}}->"I" 

10# substrate -> writer/skeptic/dreamer world-model: {{principal}}->"the user", {{self}}->assistant display name 

11# display -> human view (health, listings, Obsidian): {{principal}}->principal name, {{self}}->assistant display name 

12# 

13# {{principal}} is the canonical token (it names the role, not the pronoun, which 

14# is what lets the `audience` seam re-address it for the C-3PO trajectory). 

15# {{user}} is accepted only as a legacy back-compat synonym. 

16 

17_PRINCIPAL_TOK = re.compile(r"\{\{\s*(?:principal|user)\s*\}\}") 

18_SELF_TOK = re.compile(r"\{\{\s*self\s*\}\}") 

19_UNRESOLVED_TOK = re.compile(r"\{\{\s*[^{}]+\s*\}\}")

21_MONTH_NAMES = ( 

22 "January", "February", "March", "April", "May", "June", 

23 "July", "August", "September", "October", "November", "December", 

24) 

25_MONTH = "(?:" + "|".join(_MONTH_NAMES) + ")" 

26_DAY = r"(?:0?[1-9]|[12][0-9]|3[01])(?:st|nd|rd|th)?" 

27_YEAR = r"(?:19|20)\d{2}" 

28_DATE_PATTERNS = tuple(re.compile(pattern, re.IGNORECASE) for pattern in ( 

29 rf"\b{_MONTH}\s+{_DAY}(?:\s*,?\s*{_YEAR})?\b", 

30 rf"\b{_DAY}\s+{_MONTH}(?:\s*,?\s*{_YEAR})?\b", 

31 rf"\b{_MONTH}\s+{_YEAR}\b", 

32 rf"\b{_YEAR}[-/]{_MONTH}[-/]{_DAY}\b",

34 rf"\b(?:in|during|throughout|since|until|by)\s+{_MONTH}\b", 

35 rf"\bbirthday(?:\s+is|\s+falls)?(?:\s+sometime)?\s+in\s+{_MONTH}\b",

58def render_deixis( 

59 text: str, 

60 audience: Audience, 

61 vault: Path | None = None, 

62 *, 

63 principal_name: str | None = None, 

64) -> str: 

65 """Resolve role tokens to person for one audience. 

66 

67 ``interlocutor`` and ``substrate`` are name-independent, so ``vault`` is 

68 optional and only consulted for ``display``. For ``display`` a caller may 

69 pass ``principal_name`` directly (the fast path used by ``render_for_display``); 

70 otherwise the principal's display name is resolved from ``vault``. With 

71 neither, ``display`` falls back to "the user". 

72 """ 

73 if not text: 

74 return text 

75 if audience == "interlocutor": 

76 u, s = "you", "I" 

77 elif audience == "substrate": 

78 u, s = "the user", assistant_display_name(vault) if vault is not None else "Lisan" 

79 else: # display 

80 if principal_name: 

81 u = principal_name 

82 elif vault is not None: 

83 names = sorted(principal_aliases(vault), key=len, reverse=True) 

84 u = names[0] if names else "the user" 

85 else: 

86 u = "the user" 

87 s = assistant_display_name(vault) if vault is not None else "Lisan" 

88 text = _PRINCIPAL_TOK.sub(u, text) 

89 text = _SELF_TOK.sub(s, text) 

90 return text

93def render_obj(obj, audience: Audience, vault: Path | None = None, *, principal_name: str | None = None): 

94 """Recursively render tokens in str/list/dict structures (for narrative_state etc.).""" 

95 if isinstance(obj, str): 

96 return render_deixis(obj, audience, vault, principal_name=principal_name) 

97 if isinstance(obj, list): 

98 return [render_obj(x, audience, vault, principal_name=principal_name) for x in obj] 

99 if isinstance(obj, dict): 

100 return {k: render_obj(v, audience, vault, principal_name=principal_name) for k, v in obj.items()} 

101 return obj

119def tokenize_principal(text: str, vault: Path) -> str: 

120 """Replace the principal's literal name with the {{principal}} token. 

121 

122 Deterministic safety net for when a Writer model emits the principal's real 

123 name instead of the {{principal}} token the writer prompts request. Date 

124 expressions, quoted/verbatim text, and identifier-like occurrences are 

125 deliberately protected: a principal called August must not turn 

126 ``August 20, 2026`` into a role reference. Matches whole-word aliases (so a 

127 possessive like "Mara's" becomes "{{principal}}'s") and is idempotent. 

128 """ 

129 if not text: 

130 return text 

131 protected = _protected_spans(text) 

132 for alias in sorted((a for a in principal_aliases(vault) if a), key=len, reverse=True): 

133 # ``\b`` considers hyphens, slashes, dots, and colons boundaries, which 

134 # made names inside IDs/paths date-like identifiers eligible. These 

135 # explicit boundaries keep such machine strings literal. 

136 pattern = re.compile(rf"(?<![\w@./:#-]){re.escape(alias)}(?![\w@./:#-])") 

137 text = pattern.sub( 

138 lambda match: match.group(0) if _span_is_protected(match.span(), protected) else "{{principal}}", 

139 text, 

140 ) 

141 # Earlier substitutions change offsets. Recompute spans before the 

142 # next alias rather than applying stale coordinates. 

143 protected = _protected_spans(text) 

144 return text

147def _tokenize_principal_obj(obj, vault: Path, *, preserve_literal: bool = False): 

148 if isinstance(obj, str): 

149 return obj if preserve_literal else tokenize_principal(obj, vault) 

150 if isinstance(obj, list): 

151 return [_tokenize_principal_obj(item, vault, preserve_literal=preserve_literal) for item in obj] 

152 if isinstance(obj, dict): 

153 out = {} 

154 for key, value in obj.items(): 

155 next_preserve = preserve_literal or _literal_field(str(key)) 

156 out[key] = _tokenize_principal_obj(value, vault, preserve_literal=next_preserve) 

157 return out 

158 return obj

161def _protected_spans(text: str) -> list[tuple[int, int]]: 

162 spans: list[tuple[int, int]] = [] 

163 for pattern in (*_DATE_PATTERNS, *_QUOTED_PATTERNS): 

164 spans.extend(match.span() for match in pattern.finditer(text)) 

165 return spans

168def _span_is_protected(span: tuple[int, int], protected: list[tuple[int, int]]) -> bool: 

169 start, end = span 

170 return any(start >= protected_start and end <= protected_end for protected_start, protected_end in protected)

173def _literal_field(key: str) -> bool: 

174 """Does a structured field carry identity, source, or timestamp syntax?""" 

175 normalized = key.strip().lower().replace("-", "_") 

176 if normalized in _LITERAL_FIELD_NAMES: 

177 return True 

178 if normalized.startswith(("source_", "provenance_", "raw_", "verbatim_", "quoted_")): 

179 return True 

180 return normalized.endswith(("_id", "_ids", "_path", "_url", "_hash", "_at", "_date", "_time"))

183def has_unresolved_token(text: str) -> bool: 

184 """Return True when text still contains a raw role token.""" 

185 return bool(text and _UNRESOLVED_TOK.search(text))
