1"""Operator-only, reversible backfill of entity relationship edges. 

2 

3Until 2026-10-29 relationship edges lived only in the SQLite ``links`` 

4table and were erased by the next reindex of the entity or the nightly 

5rebuild; the live index held none. Edges now live in each source entity's 

6frontmatter ``relationships`` list. This migration writes an owner-approved 

7manifest of edges — recovered from surviving drafts' Writer output or stated 

8explicitly in entity summaries — into that list. 

9 

10The manifest (kept outside the repository, beside the bundle, because it 

11names real people) is a JSON list of ``{"source", "type", "target", 

12"basis"}``; ``"symmetric": true`` also writes the reverse edge. Every id must 

13resolve to a live entity file or prepare refuses. Existing edges are never 

14duplicated or removed. Bundle, hash-guarded apply, and rollback follow 

15story_log_repair_migration. 

16"""

18

20from datetime import datetime, timezone 

21from pathlib import Path 

22from typing import Any

24from ..frontmatter import dump_markdown, load_markdown 

25from .story_log_repair_migration import ( 

26 MigrationError, 

27 _sha, 

28 _write_json, 

29 apply_prepared_migration, 

30 rollback_migration, 

31) 

32 

33MIGRATION_ID = "20261030-relationship-backfill" 

34 

35__all__ = ["prepare_migration", "apply_prepared_migration", "rollback_migration", "MigrationError"]

38def _entity_paths(vault: Path) -> dict[str, Path]: 

39 out: dict[str, Path] = {} 

40 for path in (vault / "entities").rglob("*.md"): 

41 if "archive" in path.relative_to(vault).parts: 

42 continue 

43 try: 

44 record_id = str(load_markdown(path).frontmatter.get("id") or "") 

45 except Exception: 

46 continue 

47 if record_id: 

48 out.setdefault(record_id, path) 

49 return out 

50

52def prepare_migration(vault: Path, edges: list[dict[str, Any]], migration_root: Path) -> dict[str, Any]: 

53 if migration_root.exists(): 

54 raise MigrationError(f"migration directory already exists: {migration_root}") 

55 paths = _entity_paths(vault) 

56 expanded: list[dict[str, str]] = [] 

57 for edge in edges: 

58 source, target, rel_type = str(edge["source"]), str(edge["target"]), str(edge["type"]) 

59 for record_id in (source, target): 

60 if record_id not in paths: 

61 raise MigrationError(f"unknown entity id: {record_id}") 

62 if source == target: 

63 raise MigrationError(f"self-edge: {source}") 

64 basis = str(edge.get("basis") or "") 

65 expanded.append({"source": source, "target": target, "type": rel_type, "basis": basis}) 

66 if edge.get("symmetric"): 

67 expanded.append({"source": target, "target": source, "type": rel_type, "basis": basis}) 

68 

69 by_source: dict[str, list[dict[str, str]]] = {} 

70 for edge in expanded: 

71 by_source.setdefault(edge["source"], []).append(edge) 

72 

73 files: list[dict[str, Any]] = [] 

74 added = 0 

75 for source, new_edges in sorted(by_source.items()): 

76 path = paths[source] 

77 before = path.read_text(encoding="utf-8") 

78 doc = load_markdown(path) 

79 fm = dict(doc.frontmatter) 

80 current = [r for r in (fm.get("relationships") or []) if isinstance(r, dict)] 

81 changed = False 

82 for edge in new_edges: 

83 if any(r.get("target") == edge["target"] and r.get("type") == edge["type"] for r in current): 

84 continue 

85 current.append({"target": edge["target"], "type": edge["type"], 

86 "added_by": MIGRATION_ID, "basis": edge["basis"]}) 

87 added += 1 

88 changed = True 

89 if not changed: 

90 continue 

91 fm["relationships"] = current 

92 after = dump_markdown(fm, doc.body) 

93 rel = str(path.relative_to(vault)) 

94 for side, text in (("before", before), ("after", after)): 

95 target = migration_root / side / rel 

96 target.parent.mkdir(parents=True, exist_ok=True) 

97 target.write_text(text, encoding="utf-8") 

98 files.append({"path": rel, "mode": stat.S_IMODE(path.stat().st_mode), 

99 "before_sha256": _sha(before), "after_sha256": _sha(after)}) 

100 record = { 

101 "migration_id": MIGRATION_ID, 

102 "state": "prepared", 

103 "prepared_at": datetime.now(timezone.utc).isoformat(), 

104 "vault": str(vault), 

105 "edges_added": added, 

106 "files": files, 

107 } 

108 _write_json(migration_root / "manifest.json", {"edges": edges}) 

109 _write_json(migration_root / "migration.json", record) 

110 return record
