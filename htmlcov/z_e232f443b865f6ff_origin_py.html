1"""Where a memory came from — trust class, and source identity. 

2 

3Two different questions live here, and conflating them is how the gap this 

4module closes was created. 

5 

6**How much authority does this memory carry?** That is the *origin class*, and 

7it must be bound at the write boundary from the authenticated channel, never 

8inferred from content. Before this module, the closest thing Lisan had was 

9``reliability``, which arrived from the writer's own output 

10(``record_fanout.py``) — a language model reading possibly-adversarial content 

11and grading its own trustworthiness. The 2026 work on memory poisoning shows 

12that both content-derived and lineage-derived trust are *malleable*: an 

13adversary can transform the protected signal without forging a credential, 

14through self-summarization, trusted-tool echo, or manufactured corroboration. 

15Authority has to follow origin, and origin has to be recorded by the code that 

16receives the content. 

17 

18**Are two memories independent?** That is the *lineage key*, and it is a 

19different question with a different answer. Two ``external`` records from two 

20different documents corroborate each other. Two ``external`` records from the 

21same PDF do not — they are one source counted twice. Measured on a production 

22vault 2026-07-30: 339 of 340 knowledge records carry a ``source_document``, and 

23a single document had produced **38 chunks**. Against an evidence gate that 

24asked only ``len(support) >= 2``, that one document could found a pattern about 

25a person nineteen times over, by itself, and the result would read as 

26well-cited. 

27 

28So: class decides *how much* a memory may do; lineage decides *whether two 

29memories are really two*. 

30 

31A note on what "immutable" means here, because overclaiming it would be worse 

32than not having it. These are plain markdown files and the owner can edit any 

33field — that is the point of the architecture and it is not going to change. 

34Non-malleability is enforced against the *pipeline*: no model sets an origin, a 

35derived record cannot be more trusted than its least-trusted source, and the 

36validator fails a record that violates that. The owner remains sovereign; an 

37adversary who can only supply content does not. 

38""" 

39from __future__ import annotations 

40 

41from typing import Any, Iterable 

42 

43# --------------------------------------------------------------------------- 

44# The trust lattice 

45 

46ORIGIN_OWNER = "owner" # the principal said it, wrote it, or approved it 

47ORIGIN_AGENT = "agent" # the system's own reasoning, or its own execution 

48ORIGIN_TOOL = "tool" # output of a tool the system chose to invoke 

49ORIGIN_EXTERNAL = "external" # third-party content the system merely received 

50 

51#: Most to least trusted. Derived records inherit the **least** trusted of 

52#: their sources, which is what makes the binding non-malleable under 

53#: paraphrase: summarizing untrusted text cannot launder it into agent-authored 

54#: knowledge, because the summary's origin is computed, not written. 

55_TRUST: dict[str, int] = { 

56 ORIGIN_OWNER: 3, 

57 ORIGIN_AGENT: 2, 

58 ORIGIN_TOOL: 1, 

59 ORIGIN_EXTERNAL: 0, 

60} 

61 

62ORIGINS = frozenset(_TRUST) 

63 

64#: Unknown provenance is treated as untrusted, not as absent. A memory whose 

65#: origin cannot be established is exactly the memory that should not be able 

66#: to authorize anything, so the default fails closed. 

67DEFAULT_ORIGIN = ORIGIN_EXTERNAL 

68 

69#: The capture front door labels turns by speaker; this is the one place that 

70#: mapping lives. ``SYSTEM`` and ``ADJUTANT`` are the system acting or 

71#: reporting on itself, which is agent-authored — not owner-authored, however 

72#: much the text may read like an instruction. 

73SPEAKER_ORIGINS: dict[str, str] = { 

74 "USER": ORIGIN_OWNER, 

75 "OWNER": ORIGIN_OWNER, 

76 "LISAN": ORIGIN_AGENT, 

77 "ADJUTANT": ORIGIN_AGENT, 

78 "SYSTEM": ORIGIN_AGENT, 

79} 

80 

81 

82def trust(origin: Any) -> int: 

83 """Trust level of an origin class; unknown values are least trusted.""" 

84 return _TRUST.get(normalize_origin(origin), _TRUST[DEFAULT_ORIGIN]) 

85 

86 

87def normalize_origin(value: Any) -> str: 

88 if not isinstance(value, str): 

89 return "" 

90 text = value.strip().lower() 

91 return text if text in ORIGINS else "" 

92 

93 

94def origin_for_speaker(speaker: Any) -> str: 

95 """Origin class implied by a capture turn's speaker. 

96 

97 Anything unrecognised is ``external``: a speaker label the code does not 

98 know is content arriving from somewhere it does not control. 

99 """ 

100 if not isinstance(speaker, str): 

101 return DEFAULT_ORIGIN 

102 return SPEAKER_ORIGINS.get(speaker.strip().upper(), DEFAULT_ORIGIN) 

103 

104 

105def least_trusted(origins: Iterable[Any]) -> str: 

106 """The least trusted origin in a set — what a derived record inherits. 

107 

108 An empty set yields :data:`DEFAULT_ORIGIN`, because a record derived from 

109 nothing traceable is not thereby trustworthy. 

110 """ 

111 known = [normalize_origin(o) for o in origins] 

112 known = [o for o in known if o] 

113 if not known: 

114 return DEFAULT_ORIGIN 

115 return min(known, key=trust) 

116 

117 

118def record_origin(frontmatter: dict[str, Any]) -> str: 

119 """The origin class declared on a record, or ``""`` if it declares none. 

120 

121 Deliberately does *not* default. A caller deciding authority should use 

122 :func:`effective_origin`, which fails closed; a caller reporting on the 

123 corpus wants to know the difference between "external" and "never labelled", 

124 because every record written before this field existed is the latter. 

125 """ 

126 return normalize_origin(frontmatter.get("origin")) 

127 

128 

129def effective_origin(frontmatter: dict[str, Any]) -> str: 

130 """Origin for authority decisions: an unlabelled record is untrusted.""" 

131 return record_origin(frontmatter) or DEFAULT_ORIGIN 

132 

133 

134def may_authorize(origin: Any) -> bool: 

135 """Whether a memory of this origin may ground a consequential action. 

136 

137 Owner- and agent-authored memory may; tool output and external content may 

138 inform, never authorize. This is the elevation rule, and the corroboration 

139 path in :func:`independent_lineages` is how untrusted material earns more. 

140 """ 

141 return trust(origin) >= _TRUST[ORIGIN_AGENT] 

142 

143 

144# --------------------------------------------------------------------------- 

145# Lineage — source identity, which is a different question from trust 

146 

147#: Fields that name the *thing a record came from*, most specific first. A 

148#: document chunked into 38 records shares one ``source_document``, so grouping 

149#: on this collapses the chunks back into the single source they always were. 

150_LINEAGE_FIELDS = ( 

151 "artifact_ref", # the ingested artifact record 

152 "source_document", # the document a chunk was cut from 

153 "source_path", # the file on disk 

154 "source_uri", # where it was fetched from 

155 "batch_id", # the ingestion run 

156) 

157 

158 

159def lineage_key(frontmatter: dict[str, Any]) -> str: 

160 """A stable identity for where this record ultimately came from. 

161 

162 Falls back to the record's own id: a record with no traceable source is its 

163 own lineage, which is the conservative reading — it neither corroborates 

164 others nor is corroborated by them. 

165 """ 

166 for field_name in _LINEAGE_FIELDS: 

167 value = frontmatter.get(field_name) 

168 if isinstance(value, str) and value.strip(): 

169 return f"{field_name}:{value.strip().lower()}" 

170 record_id = frontmatter.get("id") 

171 if isinstance(record_id, str) and record_id.strip(): 

172 return f"id:{record_id.strip()}" 

173 return "" 

174 

175 

176def independent_lineages(frontmatters: Iterable[dict[str, Any]]) -> set[str]: 

177 """Distinct sources among these records. 

178 

179 This is the number an evidence gate should count. ``len(records) >= 2`` asks 

180 whether someone wrote two things down; ``len(independent_lineages) >= 2`` 

181 asks whether two sources agree, which is the question the gate was written 

182 to answer. 

183 """ 

184 keys = {lineage_key(fm) for fm in frontmatters} 

185 keys.discard("") 

186 return keys
