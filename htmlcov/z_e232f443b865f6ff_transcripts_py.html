11# The capture pipeline writes the user's turn to the transcript 

12# BEFORE the writer runs. A timeout / crash leaves an orphaned entry, and the 

13# retry then appends a second copy. We guard with a content-equality check on 

14# the most recent transcript turns: same speaker, same text, same conversation 

15# id, and within a short look-back window. 

16_DEDUP_WINDOW_MINUTES = 60

19def transcript_path_for( 

20 vault: Path | None = None, 

21 timestamp: datetime | None = None, 

22) -> Path: 

23 """Today's transcript file, created empty if absent — no appending. 

24 

25 For callers that need a transcript reference for a turn that is 

26 already in the transcript (the capture observer: the conversation 

27 layer appended both sides of the exchange at receive time).""" 

28 vault = vault or vault_root() 

29 timestamp = timestamp or datetime.now() 

30 date_str = timestamp.date().isoformat() 

31 path = vault / "transcripts" / f"{date_str}.md" 

32 path.parent.mkdir(parents=True, exist_ok=True) 

33 if not path.exists(): 

34 write_markdown(path, {"date": date_str}, "") 

35 return path

38def append_transcript( 

39 vault: Path | None = None, 

40 conversation_id: str | None = None, 

41 speaker: str = "USER", 

42 text: str = "", 

43 timestamp: datetime | None = None, 

44) -> Path: 

45 vault = vault or vault_root() 

46 timestamp = timestamp or datetime.now() 

47 date_str = timestamp.date().isoformat() 

48 time_str = timestamp.strftime("%H:%M")

51 

52 if not path.exists():

81def _is_recent_duplicate( 

82 *, 

83 existing: str, 

84 conversation_id: str | None, 

85 speaker: str, 

86 text: str, 

87 now: datetime, 

88) -> bool: 

89 """Return True if the most recent matching turn is a duplicate of this one. 

90 

91 "Matching" means: same conversation id, same speaker, identical text after 

92 whitespace normalization, written within the dedup window. We only look at 

93 the *last* turn of the same speaker in the same conversation — if the user 

94 legitimately repeats themselves later in a long session it should still 

95 capture; we're only catching back-to-back retries from a crash. 

96 """ 

97 if not text.strip(): 

98 return False 

99 normalized_new = " ".join(text.split()) 

100 cutoff = now - timedelta(minutes=_DEDUP_WINDOW_MINUTES) 

101 today_date = now.date().isoformat() 

102 

103 most_recent: tuple[datetime, str, str] | None = None 

104 for match in _BLOCK_RE.finditer("\n" + existing): 

105 block_conv = match.group("conv") or "" 

106 if (conversation_id or "") != block_conv: 

107 continue 

108 try: 

109 block_time = datetime.strptime( 

110 f"{today_date} {match.group('time')}", "%Y-%m-%d %H:%M" 

111 ) 

112 except ValueError: 

113 continue 

114 for line in match.group("body").strip().splitlines(): 

115 line = line.strip() 

116 if not line: 

117 continue 

118 head, sep, msg = line.partition(":") 

119 if not sep: 

120 continue 

121 block_speaker = head.strip() 

122 if block_speaker != speaker: 

123 continue 

124 most_recent = (block_time, block_speaker, " ".join(msg.split())) 

125 if most_recent is None: 

126 return False 

127 block_time, _, block_text = most_recent 

128 if block_time < cutoff: 

129 return False 

130 return block_text == normalized_new
