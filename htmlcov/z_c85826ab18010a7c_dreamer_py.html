15 def fallback_output(self, user_input: str, significance: str = "medium", **kwargs: Any) -> str: 

16 task = str(kwargs.get("task") or "compress") 

17 # Custom-contract tasks degrade to their honest empty answer — no 

18 # provider must never mean invented revisions or elevations. 

19 if task == "reconcile": 

20 return json.dumps({"revisions": []}) 

21 if task == "hindsight": 

22 return json.dumps({"elevations": []}) 

23 payload = { 

24 "task": task, 

25 "summary": self._summary(user_input), 

26 "findings": self._findings(task, user_input), 

27 "recommendations": self._recommendations(task, user_input), 

28 "questions": self._questions(task, user_input), 

29 "approved": task not in {"contradict", "overfitting"}, 

30 "notes": self._notes(task, user_input), 

31 } 

32 return json.dumps(payload, indent=2, ensure_ascii=True)

34 def _summary(self, text: str) -> str: 

35 first_line = next((line.strip() for line in text.splitlines() if line.strip()), "") 

36 return first_line[:140] if first_line else "Dreamer analysis"

38 def _findings(self, task: str, text: str) -> list[dict[str, str]]: 

39 lowered = text.lower() 

40 findings: list[dict[str, str]] = [] 

41 if task == "compress": 

42 findings.append({"type": "compression", "message": "Preserve claims and history while trimming operational detail."}) 

43 elif task == "primer": 

44 findings.append({"type": "primer", "message": "Primer should rely only on state, entities, recent episodes, and operating style."}) 

45 elif task == "contradict": 

46 if "but" in lowered or "however" in lowered or "yet" in lowered: 

47 findings.append({"type": "contradiction_candidate", "message": "Potential internal tension in the source text."}) 

48 elif task == "epoch": 

49 findings.append({"type": "epoch", "message": "Only propose an epoch transition for a fundamental state change."}) 

50 elif task == "confidence": 

51 findings.append({"type": "confidence", "message": "Review explicit confidence decay triggers, not intuition."}) 

52 elif task == "overfitting": 

53 findings.append({"type": "overfitting", "message": "Check for coherent stories with low evidence density."}) 

54 elif task == "identity_anchor": 

55 findings.append({"type": "identity_anchor", "message": "Surface recurring narrative patterns as hypotheses."}) 

56 if approx_word_count(text) > 120: 

57 findings.append({"type": "volume", "message": "Input is long enough to justify a summary or compression pass."}) 

58 return findings
