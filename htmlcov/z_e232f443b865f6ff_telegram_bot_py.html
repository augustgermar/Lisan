74def _telegram_api(token: str, method: str, params: dict[str, Any], *, timeout: float) -> dict[str, Any]: 

75 """Call a Telegram Bot API method. Stdlib-only; raises on transport error.""" 

76 url = _API_URL.format(token=token, method=method) 

77 data = json.dumps(params).encode("utf-8") 

78 req = urllib.request.Request(url, data=data, headers={"Content-Type": "application/json"}) 

79 with urllib.request.urlopen(req, timeout=timeout) as resp: 

80 return json.loads(resp.read().decode("utf-8"))

95def _fragment_key(update: dict[str, Any]) -> tuple[int, int] | None: 

96 """(chat_id, user_id) for a plain text message update, else None. 

97 Only these are candidates for split-message coalescing.""" 

98 message = update.get("message") 

99 if not isinstance(message, dict): 

100 return None 

101 text = message.get("text") 

102 if not isinstance(text, str) or not text.strip(): 

103 return None 

104 chat_id = (message.get("chat") or {}).get("id") 

105 user_id = (message.get("from") or {}).get("id") 

106 if chat_id is None or user_id is None: 

107 return None 

108 return int(chat_id), int(user_id)

111def _coalesce_split_messages(updates: list[dict[str, Any]]) -> list[dict[str, Any]]: 

112 """Rejoin messages the sender's Telegram client split at the 4096-char cap. 

113 

114 A long paste arrives as back-to-back fragments, each its own update; left 

115 alone, each becomes a separate turn — and an instruction like "save all of 

116 this" lands in a turn that no longer contains the "this" (2026-07-25: a 

117 three-part brief lost two parts exactly this way). A fragment is recognized 

118 by running up against the message limit: only a text message of 

119 _SPLIT_THRESHOLD chars or more absorbs the next text message from the same 

120 chat and sender. Normal rapid-fire messages stay separate turns.""" 

121 out: list[dict[str, Any]] = [] 

122 last_fragment_len = 0 

123 for update in updates: 

124 key = _fragment_key(update) 

125 if key is not None and out and _fragment_key(out[-1]) == key and last_fragment_len >= _SPLIT_THRESHOLD: 

126 text = update["message"]["text"] 

127 out[-1]["message"]["text"] += "\n" + text 

128 last_fragment_len = len(text) 

129 continue 

130 out.append(update) 

131 last_fragment_len = len(update["message"]["text"]) if key is not None else 0 

132 return out

135def _chunk(text: str, limit: int = _MSG_LIMIT) -> list[str]: 

136 """Split text into <=limit pieces, preferring newline boundaries.""" 

137 text = text.strip() 

138 if not text: 

139 return [] 

140 chunks: list[str] = [] 

141 while len(text) > limit: 

142 cut = text.rfind("\n", 0, limit) 

143 if cut <= 0: 

144 cut = limit 

145 chunks.append(text[:cut]) 

146 text = text[cut:].lstrip("\n") 

147 if text: 

148 chunks.append(text) 

149 return chunks

245 def _typing_keepalive(self, chat_id: int): 

246 """Telegram shows a chat action for only ~5 seconds; a turn takes 

247 8-25. Refresh the indicator until the reply is ready, so the pause 

248 never reads as the bot having gone quiet. Returns a stop callable.""" 

249 stop = threading.Event() 

250 

251 interval = float(getattr(self, "typing_refresh_seconds", 4.0)) 

252 

253 def _loop() -> None: 

254 while not stop.is_set(): 

255 self._typing(chat_id) 

256 stop.wait(interval) 

257 

258 thread = threading.Thread(target=_loop, daemon=True, name="typing-keepalive") 

259 thread.start() 

260 return stop.set

306 def _handle_text(self, chat_id: int, text: str) -> None: 

307 state = self._state_for(chat_id) 

308 lowered = text.lower() 

309 

310 if lowered in ("/start", "/help"): 

311 self._send_message(chat_id, _HELP_TEXT) 

312 return 

313 if lowered in ("/status", "/health"): 

314 try: 

315 self._send_message(chat_id, _status_text(self.vault, self.db_path)) 

316 except Exception as exc: 

317 log_error(self.vault, "telegram status failed", exc) 

318 self._send_message(chat_id, f"Status unavailable (it is logged): {exc}") 

319 return 

320 if lowered in ("/new", "/reset"): 

321 from .narrative_state import reset_narrative_state 

322 

323 reset_narrative_state(self.vault, state.conversation_id) 

324 state.conversation_id = f"telegram-{chat_id}-{int(time.time())}" 

325 state.advice_history = [] 

326 state.advice_context_active = False 

327 state.advice_topic = None 

328 state.trust_until = 0.0 # standing trust does not survive a conversation reset 

329 self._send_message(chat_id, "Started a fresh conversation.") 

330 return 

331 if lowered.startswith("/domain") or lowered.startswith("/arena"): 

332 parts = text.split(maxsplit=1) 

333 state.domain_override = parts[1].strip().lower() if len(parts) > 1 and parts[1].strip() else None 

334 msg = f"Domain set to: {state.domain_override}" if state.domain_override else "Domain cleared (auto-detect)." 

335 self._send_message(chat_id, msg) 

336 return 

337 if lowered.startswith("/trust"): 

338 self._send_message(chat_id, self._trust_command(state, text)) 

339 return 

340 # Adjutant confirmations: approve/deny/list, handled before the 

341 # model ever sees the text — authority decisions are deterministic. 

342 from .adjutant_confirmations import confirmation_command_response 

343 

344 confirmation_reply = confirmation_command_response(self.vault, text, db_path=self.db_path) 

345 if confirmation_reply is not None: 

346 markup = None 

347 if lowered in ("/confirmations", "/pending"): 

348 from .adjutant_confirmations import pending_confirmation_keyboard 

349 

350 markup = pending_confirmation_keyboard(self.vault, self.db_path) 

351 self._send_message(chat_id, confirmation_reply, reply_markup=markup) 

352 return 

353 if lowered.startswith("/logs"): 

354 from .log import tail_log 

355 

356 n = 20 

357 errors_only = False 

358 for part in lowered.split()[1:]: 

359 if part in ("errors", "error", "err"): 

360 errors_only = True 

361 continue 

362 try: 

363 n = int(part) 

364 except ValueError: 

365 pass 

366 self._send_message(chat_id, tail_log(self.vault, lines=n, errors_only=errors_only) or "(no logs)") 

367 return 

368 

369 # Normal turn — keep "typing" alive while the model works, then reply. 

370 stop_typing = self._typing_keepalive(chat_id) 

371 try: 

372 result = _process_chat_turn( 

373 vault=self.vault, 

374 conversation_id=state.conversation_id, 

375 text=text, 

376 provider=self.provider, 

377 model=self.model, 

378 advice_history=state.advice_history, 

379 advice_context_active=state.advice_context_active, 

380 advice_topic=state.advice_topic, 

381 domain_override=state.domain_override, 

382 db_path=self.db_path, 

383 approval_fn=self._approval_fn_for(chat_id), 

384 ) 

385 except Exception as exc: 

386 stop_typing() 

387 log_error(self.vault, "telegram turn failed", exc) 

388 self._send_message(chat_id, f"Something went wrong handling that: {exc}") 

389 return 

390 

391 stop_typing() 

392 response = str(result.get("response") or "").strip() 

393 self._update_state_after_turn(state, result, text, response) 

394 if not response: 

395 # An empty response means the pipeline swallowed an error (it only 

396 # sets result["error"] on the generic-exception path). Silence reads 

397 # as the bot ignoring the user — always say what happened instead. 

398 err = str(result.get("error") or "").strip() 

399 response = ( 

400 f"Something went wrong handling that (it's logged): {err}" 

401 if err else 

402 "I couldn't produce a response to that one — the failure is logged." 

403 ) 

404 self._send_message(chat_id, response)

464 # ── Standing trust (/trust) ───────────────────────────────────────────── 

465 _TRUST_MAX_SECONDS = 8 * 3600.0 

466 _TRUST_ARG_RE = re.compile(r"^(\d+)\s*(s|sec|secs|m|min|mins|h|hr|hrs)?$") 

467 

468 def _trust_command(self, state: _ChatState, text: str) -> str: 

469 """Session-scoped standing approval: while active, approval-gated 

470 actions run without the per-step prompt, each one announced. Expires 

471 on its own, on /trust off, and on /new — trust is a window the owner 

472 opens, never a switch that stays flipped.""" 

473 parts = text.split(maxsplit=1) 

474 arg = parts[1].strip().lower() if len(parts) > 1 else "" 

475 now = time.time() 

476 if not arg: 

477 if state.trust_until > now: 

478 minutes = int((state.trust_until - now) // 60) + 1 

479 return ( 

480 f"Standing trust is ON for another ~{minutes} min — gated actions run " 

481 "without asking (I announce each one). /trust off to end it." 

482 ) 

483 return "Standing trust is off — every gated action asks first. /trust 30m or /trust 2h to open a window." 

484 if arg in ("off", "stop", "end", "0"): 

485 state.trust_until = 0.0 

486 return "Standing trust ended — back to per-action approvals." 

487 match = self._TRUST_ARG_RE.match(arg) 

488 if not match: 

489 return "I didn't understand that window. Try /trust 30m, /trust 2h, or /trust off." 

490 value, unit = int(match.group(1)), (match.group(2) or "m") 

491 seconds = value * (3600.0 if unit.startswith("h") else 1.0 if unit.startswith("s") else 60.0) 

492 seconds = min(seconds, self._TRUST_MAX_SECONDS) 

493 state.trust_until = now + seconds 

494 get_logger(self.vault).info("telegram: standing trust granted for %ds", int(seconds)) 

495 return ( 

496 f"Standing trust ON for the next {int(seconds // 60)} min: I'll run gated actions " 

497 "without asking and tell you each time. /trust off ends it early." 

498 )

505 def _approval_fn_for(self, chat_id: int): 

506 """Ask the owner in-chat and wait for their reply. This is the 

507 approval gate, live on Telegram: the action runs only on an explicit 

508 yes, and every other message is buffered as normal conversation.""" 

509 

510 def approve(tool_name: str, args: dict[str, Any]) -> bool: 

511 import uuid 

512 

513 description = str(args.get("task") or json.dumps(args, ensure_ascii=True)[:400]) 

514 state = self._state_for(chat_id) 

515 if tool_name not in {"create_skill", "update_skill"} and time.time() < state.trust_until: 

516 get_logger(self.vault).info("telegram: %s auto-approved (standing trust): %s", tool_name, description[:200]) 

517 self._send_message(chat_id, f"⚙️ Auto-approved (standing trust): {description[:300]}") 

518 return True 

519 

520 nonce = uuid.uuid4().hex[:10] 

521 keyboard = { 

522 "inline_keyboard": [[ 

523 {"text": "✅ Approve", "callback_data": f"approve:{nonce}"}, 

524 {"text": "❌ Deny", "callback_data": f"deny:{nonce}"}, 

525 ]] 

526 } 

527 try: 

528 approval_text = f"⚙️ I need your approval to run this:\n\n{description}" 

529 # Skill proposals can be several thousand characters. Telegram 

530 # limits a message to 4096 characters, so send review text in 

531 # readable chunks and attach the approval buttons to the last. 

532 chunks = [approval_text[index:index + 3000] for index in range(0, len(approval_text), 3000)] 

533 for index, chunk in enumerate(chunks): 

534 payload = {"chat_id": chat_id, "text": chunk} 

535 if index == len(chunks) - 1: 

536 payload["reply_markup"] = keyboard 

537 self._call_api("sendMessage", payload, timeout=30) 

538 except Exception as exc: 

539 log_error(self.vault, "telegram approval prompt failed", exc) 

540 return False 

541 verdict = self._await_approval(chat_id, nonce, timeout=float(self.approval_timeout_seconds)) 

542 if verdict is None: 

543 self._send_message(chat_id, "No reply in time — skipping it. Ask again when you're ready.") 

544 return False 

545 if verdict: 

546 self._send_message(chat_id, "Approved — on it.") 

547 return True 

548 self._send_message(chat_id, "Okay — not running it.") 

549 return False 

550 

551 return approve

553 def _await_approval(self, chat_id: int, nonce: str, timeout: float) -> bool | None: 

554 """Wait for a button tap (callback query carrying our nonce) or a 

555 typed yes/no from the owner. Everything else buffers as conversation. 

556 Returns True/False on a verdict, None on timeout.""" 

557 deadline = time.monotonic() + timeout 

558 while time.monotonic() < deadline: 

559 params: dict[str, Any] = {"timeout": 10} 

560 if self._offset is not None: 

561 params["offset"] = self._offset 

562 try: 

563 payload = self._call_api("getUpdates", params, timeout=10 + _HTTP_MARGIN) 

564 except Exception: 

565 continue 

566 for update in payload.get("result") or []: 

567 self._offset = int(update["update_id"]) + 1 

568 

569 callback = update.get("callback_query") or {} 

570 if callback: 

571 sender = (callback.get("from") or {}).get("id") 

572 data = str(callback.get("data") or "") 

573 if sender is not None and self._is_allowed(int(sender)) and data.endswith(f":{nonce}"): 

574 try: 

575 self._call_api("answerCallbackQuery", {"callback_query_id": callback.get("id")}, timeout=10) 

576 except Exception: 

577 pass 

578 return data.startswith("approve:") 

579 continue # stale/foreign button tap: drop 

580 

581 message = update.get("message") or {} 

582 sender = (message.get("from") or {}).get("id") 

583 chat = (message.get("chat") or {}).get("id") 

584 text = message.get("text") 

585 if chat == chat_id and sender is not None and self._is_allowed(int(sender)) and isinstance(text, str): 

586 lowered = text.strip().lower().rstrip(".!") 

587 if lowered in self._APPROVE_WORDS: 

588 return True 

589 if lowered in self._DENY_WORDS: 

590 return False 

591 # Anything else from the owner is conversation, not a 

592 # verdict: buffer it and keep waiting for the buttons or 

593 # a yes/no. (Until 2026-07-26 any non-yes text counted as 

594 # a decline — a split-message fragment landing mid-prompt 

595 # silently vetoed the action.) 

596 self._pending_updates.append(update) 

597 continue 

598 self._pending_updates.append(update) 

599 return None

601 # ── Long-poll loop ────────────────────────────────────────────────────── 

602 def poll_once(self) -> int: 

603 """Fetch and dispatch one batch of updates. Returns the count handled.""" 

604 # Messages that arrived while we were waiting on an approval reply 

605 # were buffered; they are part of the conversation and go first. 

606 while self._pending_updates: 

607 pending, self._pending_updates = self._pending_updates, [] 

608 for update in _coalesce_split_messages(pending): 

609 try: 

610 self.handle_update(update) 

611 except Exception as exc: 

612 log_error(self.vault, "telegram pending update error", exc) 

613 params: dict[str, Any] = {"timeout": _POLL_TIMEOUT} 

614 if self._offset is not None: 

615 params["offset"] = self._offset 

616 payload = self._call_api("getUpdates", params, timeout=_POLL_TIMEOUT + _HTTP_MARGIN) 

617 updates = list(payload.get("result") or []) 

618 for update in updates: 

619 self._offset = int(update["update_id"]) + 1 

620 updates.extend(self._collect_split_continuations(updates)) 

621 for update in _coalesce_split_messages(updates): 

622 try: 

623 self.handle_update(update) 

624 except Exception as exc: # one bad update must not kill the loop 

625 log_error(self.vault, "telegram handle_update error", exc) 

626 return len(updates) 

627 

628 def _collect_split_continuations(self, updates: list[dict[str, Any]]) -> list[dict[str, Any]]: 

629 """When a batch ends in a limit-sized text fragment, the rest of the 

630 split message is still in flight — poll briefly for it so the whole 

631 instruction lands as one turn. Waiting a few seconds here is cheap 

632 next to the 8-25s turn that would otherwise run on half a message.""" 

633 tail: list[dict[str, Any]] = [] 

634 deadline = time.monotonic() + _SPLIT_FOLLOWUP_SECONDS 

635 while time.monotonic() < deadline: 

636 batch = tail or updates 

637 if not batch: 

638 break 

639 last = batch[-1] 

640 if _fragment_key(last) is None or len(last["message"]["text"]) < _SPLIT_THRESHOLD: 

641 break 

642 params: dict[str, Any] = {"timeout": 2} 

643 if self._offset is not None: 

644 params["offset"] = self._offset 

645 try: 

646 payload = self._call_api("getUpdates", params, timeout=2 + _HTTP_MARGIN) 

647 except Exception: 

648 break # transport hiccup: dispatch what we have 

649 for update in payload.get("result") or []: 

650 self._offset = int(update["update_id"]) + 1 

651 tail.append(update) 

652 return tail 

653 

654 def run(self, *, _forever: bool = True, _on_idle: Callable[[], None] | None = None) -> int: 

655 backoff = 1 

656 while True: 

657 try: 

658 self.poll_once() 

659 backoff = 1 

660 except KeyboardInterrupt: 

661 return 0 

662 except Exception as exc: 

663 if _is_poll_timeout(exc): 

664 # Routine long-poll churn: the connection outlived its 

665 # window with nothing to say. One WARN line — a full 

666 # traceback 15-20x a day buried real errors — and no 

667 # backoff: backing off here made the bot deaf for up to 

668 # a minute per lull. 

669 get_logger(self.vault).warning("telegram long-poll timeout; reconnecting") 

670 elif isinstance(exc, urllib.error.URLError): 

671 log_error(self.vault, f"telegram poll network error; retry in {backoff}s", exc) 

672 time.sleep(backoff) 

673 backoff = min(backoff * 2, 60) 

674 else: 

675 log_error(self.vault, f"telegram poll error; retry in {backoff}s", exc) 

676 time.sleep(backoff) 

677 backoff = min(backoff * 2, 60) 

678 if not _forever: 

679 if _on_idle: 

680 _on_idle() 

681 return 0

684def _is_poll_timeout(exc: BaseException) -> bool: 

685 """A read timeout on the getUpdates long poll — expected, not an error.""" 

686 if isinstance(exc, TimeoutError): 

687 return True 

688 if isinstance(exc, urllib.error.URLError) and isinstance(getattr(exc, "reason", None), TimeoutError): 

689 return True 

690 return False

717def _resolve_settings(config: dict[str, Any], *, include_env: bool = True) -> tuple[str, set[int]]: 

718 """Token + allowlist from env, the credentials store, or the config block. 

719 

720 ``include_env=False`` resolves without the environment — used to check what 

721 a detached service (which never inherits this shell's env) will actually 

722 see. The credentials file IS visible to a detached service, so it is 

723 consulted in both modes. 

724 

725 Precedence: environment, then ``<credentials_root>/telegram.json``, then 

726 ``config.telegram.token``. The config fallback is deliberate and stays: 

727 this token used to live there, `backup.create_backup` copies config.json 

728 into every archive, and an install that has not migrated must keep working 

729 rather than silently losing its bot. `lisan telegram setup` writes to the 

730 credentials store now, so migration happens by using the tool. 

731 """ 

732 tg_cfg = config.get("telegram", {}) if isinstance(config.get("telegram"), dict) else {} 

733 

734 env_token = os.environ.get("LISAN_TELEGRAM_TOKEN") if include_env else None 

735 token = (env_token or _token_from_credentials_file() or str(tg_cfg.get("token") or "")).strip() 

736 

737 raw_allowed = os.environ.get("LISAN_TELEGRAM_ALLOWED") if include_env else None 

738 if raw_allowed is None: 

739 raw_allowed = tg_cfg.get("allowed_user_ids") or "" 

740 if isinstance(raw_allowed, (list, tuple)): 

741 allowed_iter = [str(x) for x in raw_allowed] 

742 else: 

743 allowed_iter = str(raw_allowed).split(",") 

744 allowed = {int(x.strip()) for x in allowed_iter if str(x).strip().lstrip("-").isdigit()} 

745 

746 return token, allowed

881def detect_owner_id( 

882 token: str, 

883 *, 

884 api: ApiFn = _telegram_api, 

885 max_wait: float = 120.0, 

886 on_wait: Callable[[], None] | None = None, 

887) -> tuple[int, str] | None: 

888 """Poll getUpdates until someone messages the bot; return (user_id, display_name). 

889

892 offset: int | None = None

903 offset = int(update["update_id"]) + 1 

904 message = update.get("message") or update.get("edited_message") or {} 

905 sender = message.get("from") or {} 

906 if sender.get("id"): 

907 name = str(sender.get("first_name") or sender.get("username") or "") 

908 return int(sender["id"]), name 

909 if time.monotonic() >= deadline: 

910 return None 

911 if on_wait: 

912 on_wait()

915def save_telegram_settings(token: str, allowed_ids: list[int], *, path: Path | None = None) -> Path: 

916 """Persist the token to the credentials store and the allowlist to config. 

917 

918 The token used to live in ``config.telegram.token``. That file is gitignored, 

919 but ``backup.create_backup`` copies it into every archive alongside the 

920 vault, unencrypted by default — so a routine backup published a credential 

921 that grants full control of the running agent to anyone holding it. One was 

922 found on disk at mode 644 on 2026-08-13. 

923 

924 The token now goes to ``<credentials_root>/telegram.json`` at mode 0600, 

925 outside both the repo and the vault, and any token already in config.json is 

926 removed as a side effect: calling this is the migration. The allowlist stays 

927 in config — it is authorization, not a secret, and keeping it there means a 

928 failure to read the credentials file cannot silently widen who may talk to 

929 the bot. 

930 

931 Returns the credentials path, which is what the operator needs to know. 

932 """ 

933 from ..paths import config_path 

934 

935 path = path or config_path() 

936 cfg: dict[str, Any] = {} 

937 if path.exists(): 

938 try: 

939 cfg = json.loads(path.read_text(encoding="utf-8") or "{}") 

940 except Exception: 

941 cfg = {} 

942 if not cfg: 

943 from copy import deepcopy 

944 

945 from ..config import DEFAULT_CONFIG 

946 

947 cfg = deepcopy(DEFAULT_CONFIG) 

948 

949 cred_path = telegram_credentials_path() 

950 cred_path.parent.mkdir(parents=True, exist_ok=True) 

951 # Create with owner-only permissions rather than writing then chmod-ing: 

952 # between those two calls the token is world-readable on a shared machine. 

953 fd = os.open(cred_path, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600) 

954 with os.fdopen(fd, "w", encoding="utf-8") as handle: 

955 json.dump({"token": token}, handle, indent=2) 

956 handle.write("\n") 

957 os.chmod(cred_path, 0o600) 

958 

959 # The allowlist stays; the token does not. Leaving it would defeat the move, 

960 # since _resolve_settings falls back to config for un-migrated installs. 

961 cfg["telegram"] = {"allowed_user_ids": list(allowed_ids)} 

962 path.parent.mkdir(parents=True, exist_ok=True) 

963 path.write_text(json.dumps(cfg, indent=2) + "\n", encoding="utf-8") 

964 return cred_path

967def run_telegram_setup() -> int: 

968 """Interactive wizard: validate a bot token, auto-detect your user id, save config.""" 

969 import getpass 

970 

971 config = load_config() 

972 existing = config.get("telegram", {}) if isinstance(config.get("telegram"), dict) else {} 

973 

974 print("⚕ Lisan — Telegram setup\n") 

975 if existing.get("token"): 

976 if input("Telegram is already configured. Reconfigure? [y/N] ").strip().lower() not in ("y", "yes"): 

977 return 0 

978 print() 

979 

980 print("First, create a bot:") 

981 print(" 1. Open Telegram and message @BotFather") 

982 print(" 2. Send /newbot and follow the prompts") 

983 print(" 3. Copy the token it gives you\n") 

984

986 token = "" 

987 while True:

990 print("Aborted.") 

991 return 1 

992 if not _valid_token_format(entered): 

993 print(" That doesn't look like a token (expected <digits>:<hash>). Try again.\n") 

994 continue 

995 me = get_me(entered) 

996 if not me: 

997 print(" Telegram rejected that token. Double-check it and try again.\n") 

998 continue 

999 token = entered 

1000 print(f" ✓ Connected to @{me.get('username')} ({me.get('first_name', '')})\n") 

1001 break 

1002 

1003 allowed: list[int] = [] 

1004 print("Now let's authorize your account (so only you can use the bot):") 

1005 print(f" → Open Telegram, find @{me.get('username')}, and send it any message (e.g. 'hi').") 

1006 print(" Waiting for your message… (Ctrl-C to enter your id manually)") 

1007 try: 

1008 detected = detect_owner_id(token) 

1009 except KeyboardInterrupt: 

1010 detected = None 

1011 print() 

1012 if detected: 

1013 uid, name = detected 

1014 print(f" ✓ Got it — {name or 'you'} (id {uid})") 

1015 allowed.append(uid) 

1016 else: 

1017 manual = input(" Enter your numeric Telegram user id: ").strip() 

1018 if manual.lstrip("-").isdigit(): 

1019 allowed.append(int(manual)) 

1020 

1021 extra = input(" Additional allowed user ids (comma-separated, optional): ").strip() 

1022 for part in extra.split(","): 

1023 part = part.strip() 

1024 if part.lstrip("-").isdigit() and int(part) not in allowed: 

1025 allowed.append(int(part)) 

1026 

1027 if not allowed: 

1028 print("\n✗ No users authorized — aborting (the bot would refuse everyone).") 

1029 return 1

1032 if not set_my_commands(token): 

1033 print(" ⚠ Telegram's command menu could not be updated; the bot will retry when it starts.") 

1034 print(f"\n✓ Token saved to {path} (mode 0600, outside the repo and the vault).") 

1035 print(" It is deliberately not in config.json: backups copy that file.") 

1036 print(f" Authorized ids: {', '.join(map(str, allowed))}") 

1037 print("\nStart the bot with: lisan telegram run") 

1038 print("Keep it always-on with: lisan telegram install-service") 

1039 return 0

1117def install_service(*, vault: Path | None = None) -> int: 

1118 """Install + start the Telegram bot as an always-on OS service.""" 

1119 vault = vault or vault_root() 

1120 config = load_config() 

1121 # The service runs detached from this shell, so it only sees config.json — 

1122 # env-only settings would pass a naive check here and then crash-loop under 

1123 # launchd/systemd. Validate against what the service will actually see. 

1124 token, allowed = _resolve_settings(config, include_env=False) 

1125 if not token or not allowed: 

1126 env_token, env_allowed = _resolve_settings(config) 

1127 if env_token and env_allowed: 

1128 print("✗ Your Telegram settings are only in shell environment variables, which") 

1129 print(" the always-on service won't inherit. Run `lisan telegram setup` to") 

1130 print(" save them to config.json, then re-run install-service.") 

1131 else: 

1132 print("✗ Configure the bot first: run `lisan telegram setup`.") 

1133 return 1 

1134 

1135 system = platform.system() 

1136 if system == "Darwin": 

1137 return _install_launchd(vault) 

1138 if system == "Linux": 

1139 return _install_systemd(vault) 

1140 print(f"✗ Automatic service install isn't supported on {system}.") 

1141 print(" Run `lisan telegram run` under your own process manager.") 

1142 return 1
