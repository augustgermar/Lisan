1"""Build an evidence-gated manifest for the August/month deixis defect. 

2 

3This module is intentionally dry-run only. It can describe candidate edits, 

4but it contains no apply path: owner review is a hard boundary between finding 

5legacy damage and changing durable memory. 

6""" 

7from __future__ import annotations 

8 

9import json 

10import re 

11from collections import defaultdict 

12from dataclasses import asdict, dataclass 

13from pathlib import Path 

14

16TOKEN = r"\{\{\s*principal\s*\}\}" 

17DATE_CANDIDATE = re.compile( 

18 rf"{TOKEN}\s+(?:(?P<day>3[01]|[12][0-9]|0?[1-9])(?P<ordinal>st|nd|rd|th)?(?!\d)" 

19 rf"(?P<year_part>\s*,?\s*(?P<year>(?:19|20)\d{{2}}))?|(?P<year_only>(?:19|20)\d{{2}}))", 

20 re.IGNORECASE, 

21) 

22MONTH_ONLY_CANDIDATE = re.compile( 

23 rf"(?P<prefix>(?:birthday(?:\s+is|\s+falls)?(?:\s+sometime)?\s+in|sometime\s+in)\s+){TOKEN}", 

24 re.IGNORECASE, 

25) 

26TEXT_SUFFIXES = {".md", ".json", ".jsonl", ".yaml", ".yml", ".txt"} 

27WORD = re.compile(r"[a-z0-9]{3,}") 

28STOP = { 

29 "the", "and", "for", "that", "this", "with", "from", "into", "was", "were", "has", "had", 

30 "user", "principal", "august", "2026", "2024", "2015", "2012", "record", "reported", 

31}

37 line: int 

38 column: int 

39 corrupted_value: str 

40 proposed_value: str | None

42 evidence_source: str | None 

43 evidence_kind: str | None 

44 evidence_excerpt: str | None 

45 status: str

48def build_manifest(vault: Path) -> list[RepairCandidate]: 

49 evidence_lines = _evidence_lines(vault) 

50 candidates: list[RepairCandidate] = [] 

51 for path in _text_files(vault): 

52 text = path.read_text(encoding="utf-8", errors="replace") 

53 matches = list(DATE_CANDIDATE.finditer(text)) + list(MONTH_ONLY_CANDIDATE.finditer(text)) 

54 matches.sort(key=lambda match: match.start()) 

55 for match in matches: 

56 token_match = re.search(TOKEN, match.group(0), re.IGNORECASE) 

57 if token_match is None: 

58 continue 

59 start = match.start() + token_match.start() 

60 end = match.end() 

61 corrupted = text[start:end] 

62 corrected = re.sub(TOKEN, "August", corrupted, count=1, flags=re.IGNORECASE) 

63 context = text[max(0, start - 220):min(len(text), end + 220)] 

64 evidence = _find_evidence( 

65 vault=vault, 

66 source_path=path, 

67 corrected=corrected, 

68 context=context, 

69 evidence_lines=evidence_lines, 

70 ) 

71 line = text.count("\n", 0, start) + 1 

72 line_start = text.rfind("\n", 0, start) + 1 

73 accepted_evidence = bool(evidence and evidence[1] in { 

74 "original transcript", "structured ISO timestamp", 

75 }) 

76 status = "supported" if accepted_evidence else "unresolved" 

77 candidates.append(RepairCandidate( 

78 path=str(path.relative_to(vault)), 

79 line=line, 

80 column=start - line_start + 1, 

81 corrupted_value=corrupted, 

82 proposed_value=corrected if accepted_evidence else None, 

83 occurrence_sha256_input=f"{path.relative_to(vault)}:{start}:{corrupted}", 

84 evidence_source=evidence[0] if evidence else None, 

85 evidence_kind=evidence[1] if evidence else None, 

86 evidence_excerpt=evidence[2] if evidence else None, 

87 status=status, 

88 )) 

89 return candidates

92def write_manifest(vault: Path, json_path: Path, markdown_path: Path) -> tuple[int, int, int]: 

93 candidates = build_manifest(vault) 

94 payload = { 

95 "version": 1, 

96 "mode": "dry-run-only", 

97 "vault": str(vault), 

98 "replacement_rule": "Only evidence-supported date/month uses of {{principal}} may become August.", 

99 "candidates": [asdict(candidate) for candidate in candidates], 

100 } 

101 json_path.parent.mkdir(parents=True, exist_ok=True) 

102 json_path.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8") 

103 markdown_path.write_text(_render_markdown(candidates, json_path), encoding="utf-8") 

104 return len(candidates), sum(c.status == "supported" for c in candidates), len({c.path for c in candidates})

107def _text_files(vault: Path) -> list[Path]: 

108 return sorted( 

109 path for path in vault.rglob("*") 

110 if path.is_file() and path.suffix.lower() in TEXT_SUFFIXES

114def _evidence_lines(vault: Path) -> list[tuple[Path, int, str, str]]: 

115 lines: list[tuple[Path, int, str, str]] = [] 

116 for directory, kind in (("transcripts", "original transcript"), ("knowledge", "imported source")): 

117 root = vault / directory 

118 if not root.exists(): 

119 continue 

120 for path in sorted(root.rglob("*.md")): 

121 for number, line in enumerate(path.read_text(encoding="utf-8", errors="replace").splitlines(), start=1): 

122 if line.strip() and (kind == "original transcript" or "August" in line): 

123 lines.append((path, number, line, kind)) 

124 return lines

127def _find_evidence( 

128 *, 

129 vault: Path, 

130 source_path: Path, 

131 corrected: str, 

132 context: str, 

133 evidence_lines: list[tuple[Path, int, str, str]], 

134) -> tuple[str, str, str] | None: 

135 corrected_norm = _normalize(corrected) 

136 context_words = set(WORD.findall(_normalize(context))) - STOP 

137 date_parts = _date_parts(corrected) 

138 

139 # Prefer an ISO timestamp in the affected record itself over a merely 

140 # coincident same-day sentence elsewhere. This is the strongest link for 

141 # generated entity narratives whose source-log entry carries the date. 

142 if date_parts: 

143 year, day = date_parts 

144 iso = f"{year}-08-{day:02d}" 

145 source_text = source_path.read_text(encoding="utf-8", errors="replace") 

146 if iso in source_text: 

147 line = source_text[:source_text.index(iso)].count("\n") + 1 

148 return str(source_path.relative_to(vault)) + f":{line}", "structured ISO timestamp", iso 

149 else: 

150 year_only = re.fullmatch(r"August\s+((?:19|20)\d{2})", corrected, re.IGNORECASE) 

151 if year_only: 

152 year = int(year_only.group(1)) 

153 source_text = source_path.read_text(encoding="utf-8", errors="replace") 

154 iso_match = re.search(rf"\b{year}-08-(?:0[1-9]|[12][0-9]|3[01])\b", source_text) 

155 if iso_match: 

156 line = source_text[:iso_match.start()].count("\n") + 1 

157 return ( 

158 str(source_path.relative_to(vault)) + f":{line}", 

159 "structured ISO timestamp", 

160 iso_match.group(0), 

161 ) 

162 

163 ranked: list[tuple[int, Path, int, str, str]] = [] 

164 for path, number, line, kind in evidence_lines: 

165 normalized = _normalize(line) 

166 line_words = set(WORD.findall(normalized)) - STOP 

167 overlap = context_words & line_words 

168 same_dated_transcript = False 

169 if date_parts and kind == "original transcript": 

170 year, day = date_parts 

171 same_dated_transcript = path.stem == f"{year}-08-{day:02d}" 

172 score = 0 

173 exact = bool(corrected_norm and re.search( 

174 rf"(?<![a-z0-9]){re.escape(corrected_norm)}(?![a-z0-9])", 

175 normalized, 

176 )) 

177 if exact: 

178 score += 120 

179 elif date_parts and _contains_date(normalized, *date_parts): 

180 score += 80 

181 elif corrected.lower() == "august" and "birthday" in normalized and "august" in normalized: 

182 score += 70 

183 elif same_dated_transcript and overlap: 

184 score += 70 

185 else: 

186 continue 

187 if date_parts and not same_dated_transcript and len(overlap) < 2 and not (exact and overlap): 

188 continue 

189 if corrected.lower() == "august" and "birthday" not in line.lower(): 

190 continue 

191 score += min(25, len(overlap) * 3) 

192 if same_dated_transcript: 

193 score += 35 

194 filename_words = set(WORD.findall(path.stem.lower())) - STOP 

195 score += min(12, len(context_words & filename_words) * 3) 

196 score += 80 if kind == "original transcript" else 15 

197 ranked.append((score, path, number, line, kind)) 

198 if ranked: 

199 _, path, number, line, kind = max(ranked, key=lambda item: (item[0], -len(item[3]))) 

200 return str(path.relative_to(vault)) + f":{number}", kind, _clip(line) 

201 

202 return None

205def _date_parts(value: str) -> tuple[int, int] | None: 

206 match = re.search( 

207 r"\bAugust\s+(3[01]|[12]\d|0?[1-9])(?:st|nd|rd|th)?(?!\d)(?:\s*,?\s*((?:19|20)\d{2}))?", 

208 value, 

209 re.IGNORECASE, 

210 ) 

211 if not match: 

212 return None 

213 return int(match.group(2) or 2026), int(match.group(1))

216def _contains_date(normalized: str, year: int, day: int) -> bool: 

217 pattern = rf"\baugust\s+0?{day}(?:st|nd|rd|th)?\b" 

218 if not re.search(pattern, normalized): 

219 return False 

220 # A transcript filename supplies 2026 implicitly; literal years, when 

221 # present in source text, must agree. 

222 years = {int(value) for value in re.findall(r"\b(?:19|20)\d{2}\b", normalized)} 

223 return not years or year in years

226def _normalize(value: str) -> str: 

227 value = value.lower() 

228 value = re.sub(r"(\d)(?:st|nd|rd|th)\b", r"\1", value) 

229 return " ".join(re.findall(r"[a-z0-9]+", value))

232def _clip(value: str, limit: int = 180) -> str: 

233 value = " ".join(value.split()) 

234 return value if len(value) <= limit else value[:limit - 1] + "…"

237def _escape(value: str | None) -> str: 

238 if value is None: 

239 return "—" 

240 return value.replace("|", "\\|").replace("\n", " ")

243def _render_markdown(candidates: list[RepairCandidate], json_path: Path) -> str: 

244 grouped: dict[tuple[str, str, str | None, str | None], list[RepairCandidate]] = defaultdict(list) 

245 for candidate in candidates: 

246 key = ( 

247 candidate.path, 

248 candidate.corrupted_value, 

249 candidate.proposed_value, 

250 candidate.evidence_source, 

251 ) 

252 grouped[key].append(candidate) 

253 supported = sum(candidate.status == "supported" for candidate in candidates) 

254 unresolved = len(candidates) - supported 

255 files = len({candidate.path for candidate in candidates}) 

256 lines = [ 

257 "# Deixis/date repair manifest — dry run", 

258 "", 

259 "> No live record has been changed. A missing proposal means the month could not be independently evidenced and must not be guessed.", 

260 "", 

261 "- mode: **dry-run-only**", 

262 f"- affected records: **{files}**", 

263 f"- candidate occurrences: **{len(candidates)}**", 

264 f"- evidence-supported: **{supported}**", 

265 f"- unresolved: **{unresolved}**", 

266 f"- machine-readable manifest: `{json_path}`", 

267 "", 

268 "| Record | Location | Count | Corrupted value | Proposed corrected value | Evidence source | Evidence kind |", 

269 "|---|---:|---:|---|---|---|---|", 

270 ] 

271 ordered = sorted( 

272 grouped.items(), 

273 key=lambda item: tuple("" if value is None else value for value in item[0]), 

274 ) 

275 for (path, corrupted, proposed, evidence_source), members in ordered: 

276 locations = ", ".join(f"L{member.line}:C{member.column}" for member in members) 

277 evidence_kind = members[0].evidence_kind 

278 lines.append( 

279 f"| `{_escape(path)}` | {_escape(locations)} | {len(members)} | " 

280 f"`{_escape(corrupted)}` | `{_escape(proposed)}` | " 

281 f"`{_escape(evidence_source)}` | {_escape(evidence_kind)} |" 

282 ) 

283 lines.extend(( 

284 "", 

285 "## Rules used", 

286 "", 

287 "1. Candidate detection is limited to date-shaped role tokens and explicit birthday-month grammar.", 

288 "2. The proposed month is emitted only when supported by an original transcript, an imported literal source, or an exact same-record ISO date.", 

289 "3. The tool has no apply mode. Approval and a separate reversible migration are required.", 

290 "4. Archived records and drafts are included because they remain queryable or can seed later derived records.", 

291 "", 

292 )) 

293 return "\n".join(lines)
