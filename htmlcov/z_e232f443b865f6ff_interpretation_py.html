1"""IIP Phase 1 — the Interpersonal Interpretation Protocol. 

2 

3The finding this closes (owner + external review, 2026-07-15): asked why 

4someone kept repeating a question, the agent produced three readings that 

5all lived inside one person's psychology — no hypothesis where the user 

6was a causal factor, no boring base-rate reading, no discriminating 

7observations, no convergent action. The system's blind spots correlate 

8with the owner's, because the narrative store is its training data; a 

9hypothesis space drawn only from the corpus faithfully reproduces the 

10corpus's own priors. 

11 

12Three pieces, WO-GROUND's proven shape (deterministic detector → injected 

13directive → the model still writes the reply → deterministic validation): 

14 

15- ``is_interpretation_query`` — regex detector for turns asking to 

16 interpret another person's behavior, motives, or mental state. Runs 

17 every turn; over-inclusive by design; self-questions and system 

18 questions are excluded (they have their own machinery). 

19- ``INTERPRETATION_DIRECTIVE`` — the injected protocol block requiring a 

20 structured hypothesis space alongside the prose. 

21- ``validate_interpretation`` — deterministic post-hoc check: locus 

22 diversity (≥1 user_causal, ≥1 situational_baserate), non-empty 

23 discriminators, a convergent action (or the explicit word "none"), 

24 and provenance refs that resolve when present. Empty provenance is 

25 VALID by owner decree — an out-of-register hypothesis (Phase 2) has, 

26 by definition, nothing in the corpus to cite. 

27 

28Every detector fire is logged to ``vault/logs/iip-challenges.jsonl`` 

29(owner decree: detector precision must be visible, not only challenges). 

30The log carries a query digest, never the query text. 

31"""

52# A person is in play: third-person pronouns, kin/relation words, or a 

53# capitalized mid-sentence token (a name). Deliberately loose. 

54_PERSON_RE = re.compile( 

55 r"\b(she|he|they|her|him|them|hers|his|theirs)\b" 

56 r"|\bmy (wife|husband|ex|co[- ]?parent|mother|father|mom|dad|daughters?|sons?|kids?|" 

57 r"girls?|boys?|sisters?|brothers?|friends?|boss|in[- ]laws?|mother[- ]in[- ]law|neighbor\w*)\b" 

58) 

59 

60_NAME_RE = re.compile(r"(?<!^)(?<![.!?] )\b[A-Z][a-z]{2,}\b") # capitalized, not sentence-start 

61 

62_ASK_RE = re.compile( 

63 r"\bwhy (does|do|did|is|are|was|were|would|won.?t|can.?t|keeps?|keep)\b" 

64 r"|\bwhat (do you|should i|do we) make of\b" 

65 r"|\bhow (should|do|would) i (read|take|interpret|respond to|handle|hear)\b" 

66 r"|\bhelp me (read|understand|interpret|decode|make sense of)\b" 

67 r"|\bdecode\b" 

68 r"|\bwhat (did|does|do) (she|he|they|\w+) (mean|want|really)\b" 

69 r"|\bkeeps? (asking|saying|doing|texting|sending|bringing up)\b" 

70 r"|\bwhat.?s (going on|the deal|up) with (her|him|them)\b" 

71 r"|\b(is|are) (she|he|they) (trying|hoping|angling|hinting)\b" 

72)

75def is_interpretation_query(text: str) -> bool: 

76 """Deterministic, every turn, no LLM. True when the turn asks for a 

77 reading of another person's behavior/motives/state. 

78 

79 Requires a THIRD-PERSON signal: bare "you"/"your" never counts, so 

80 asks aimed at the agent itself ("why do you keep failing?") fall 

81 through to the self-question machinery instead. A turn naming both a 

82 person and system plumbing ("why does Ruth keep asking about my 

83 reminder system?") is an interpersonal read — and may legitimately 

84 carry the self-state block too; the two injections are independent.""" 

85 raw = str(text or "").strip() 

86 if not raw: 

87 return False 

88 lowered = " ".join(raw.lower().split()) 

89 if not _ASK_RE.search(lowered): 

90 return False 

91 return bool(_PERSON_RE.search(lowered) or _NAME_RE.search(raw))

94# ── The injected protocol ────────────────────────────────────────────────────

96INTERPRETATION_DIRECTIVE = """\ 

97This turn asks you to interpret another person's behavior. Alongside your 

98prose response, your JSON output MUST include an "interpretation" object: 

99 

100 "interpretation": { 

101 "hypotheses": [{"text": "...", "locus": "...", "provenance": ["record ids"]}], 

102 "discriminators": ["observable data that would tell the candidates apart"], 

103 "convergent_action": "an action sound under all leading hypotheses, or the word none with why" 

104 } 

105 

106Rules, all hard: 

107- locus is one of: other_person, user_causal, situational_baserate, 

108 relationship_system. Include AT LEAST ONE user_causal hypothesis (the 

109 user's own words, actions, or ambivalence as a causal factor) and AT 

110 LEAST ONE situational_baserate hypothesis — boring by design: 

111 developmental stage, disability-related pattern, logistics, ordinary 

112 planning behavior; explanations requiring no psychological modeling of 

113 anyone. 

114- Order hypotheses by prior probability, boring first unless the record 

115 says otherwise. 

116- provenance lists the stored record ids that informed a hypothesis. 

117 An empty list is legitimate — some hypotheses SHOULD come from outside 

118 the record. Never invent ids. 

119- discriminators: what to look for, ask, or check that would distinguish 

120 the top candidates. At least one. 

121- convergent_action: if one action works under all leading readings, say 

122 it prominently in the prose too. If none exists, write "none" and say 

123 what decision actually forks. 

124- The prose stays natural and readable and must agree with the structure — 

125 the object is scaffolding, not the reply."""

130def _ref_resolves(ref: str, db_path: Path | None) -> bool: 

131 """True when a provenance ref names a real indexed record. On any 

132 infrastructure trouble, err permissive — the validator must never 

133 fail a turn because the index was busy.""" 

134 try: 

135 from .db import connect as _db_connect 

136 

137 conn = _db_connect(db_path or sqlite_path(), readonly=True) 

138 try: 

139 row = conn.execute("SELECT 1 FROM files WHERE id = ? LIMIT 1", (str(ref),)).fetchone() 

140 return row is not None 

141 finally: 

142 conn.close() 

143 except Exception: 

144 return True

147def validate_interpretation(payload: dict[str, Any], *, db_path: Path | None = None) -> list[str]: 

148 """Deterministic structural check. Returns complaints; empty = pass.""" 

149 complaints: list[str] = [] 

150 interp = payload.get("interpretation") 

151 if not isinstance(interp, dict): 

152 return ["missing the required interpretation object"] 

153 hypotheses = [h for h in (interp.get("hypotheses") or []) if isinstance(h, dict)] 

154 if not hypotheses: 

155 return ["interpretation.hypotheses is empty"] 

156 loci = [str(h.get("locus") or "") for h in hypotheses] 

157 for locus in set(loci) - LOCI: 

158 complaints.append(f"invalid locus {locus!r} (allowed: {', '.join(sorted(LOCI))})") 

159 for required in sorted(REQUIRED_LOCI - set(loci)): 

160 complaints.append(f"no {required} hypothesis present") 

161 if not [d for d in (interp.get("discriminators") or []) if str(d).strip()]: 

162 complaints.append("no discriminators — name observable data that would distinguish the candidates") 

163 if not str(interp.get("convergent_action") or "").strip(): 

164 complaints.append('convergent_action missing — state one, or the word "none" with what forks') 

165 # Provenance: [] is valid (owner decree); refs must resolve when given. 

166 for h in hypotheses: 

167 for ref in (h.get("provenance") or []): 

168 if not _ref_resolves(str(ref), db_path): 

169 complaints.append(f"provenance ref {str(ref)!r} does not resolve to a stored record") 

170 return complaints

173def incompleteness_notice(complaints: list[str]) -> str: 

174 """The user-facing note when regeneration is exhausted. Honest and 

175 specific: says which part of the hypothesis space is missing.""" 

176 missing = [] 

177 joined = " ".join(complaints) 

178 if "user_causal" in joined: 

179 missing.append("a reading where you are a causal factor") 

180 if "situational_baserate" in joined: 

181 missing.append("a boring base-rate reading") 

182 if "discriminators" in joined: 

183 missing.append("what to watch for that would settle it") 

184 if "convergent_action" in joined: 

185 missing.append("an action that holds under all readings") 

186 detail = "; ".join(missing) or "part of the required hypothesis space" 

187 return f"(A note on my own reasoning: this read may be incomplete — I couldn't produce {detail}.)"

192def log_iip_event(vault: Path | None, event: dict[str, Any]) -> None: 

193 """One JSONL line per detector fire. Best-effort: telemetry never 

194 fails the turn it describes. Query text never lands here — digest only.""" 

195 try: 

196 vault = vault or vault_root() 

197 path = vault / "logs" / LOG_NAME 

198 path.parent.mkdir(parents=True, exist_ok=True) 

199 record = {"ts": datetime.now(timezone.utc).isoformat(timespec="seconds"), **event} 

200 with path.open("a", encoding="utf-8") as fh: 

201 fh.write(json.dumps(record, ensure_ascii=True) + "\n") 

202 except Exception: 

203 pass

206def query_digest(text: str) -> str: 

207 return hashlib.sha1(str(text or "").encode("utf-8")).hexdigest()[:12]

210def summarize_iip_log(vault: Path | None = None, *, weeks: int = 4) -> str: 

211 """Weekly counts from the fire/challenge log — the instrument for 

212 'is the system inheriting the owner's blind spots' and for judging 

213 detector precision and the regeneration cap. Plain text, per week.""" 

214 from collections import defaultdict 

215 from datetime import timedelta 

216 

217 vault = vault or vault_root() 

218 path = vault / "logs" / LOG_NAME 

219 if not path.exists(): 

220 return "No IIP events logged yet." 

221 cutoff = (datetime.now(timezone.utc) - timedelta(weeks=weeks)).isoformat() 

222 buckets: dict[str, dict[str, int]] = defaultdict(lambda: defaultdict(int)) 

223 for line in path.read_text(encoding="utf-8").splitlines(): 

224 try: 

225 event = json.loads(line) 

226 except json.JSONDecodeError: 

227 continue 

228 ts = str(event.get("ts") or "") 

229 if ts < cutoff: 

230 continue 

231 week = datetime.fromisoformat(ts).strftime("%G-W%V") if ts else "unknown" 

232 b = buckets[week] 

233 b["fires"] += 1 

234 b[f"validated_{event.get('validated', 'unknown')}"] += 1 

235 if event.get("regenerations"): 

236 b["regenerated"] += int(event["regenerations"]) 

237 challenge = event.get("challenge") 

238 if isinstance(challenge, dict): 

239 b["challenges"] += 1 

240 b[f"challenge_{challenge.get('outcome', 'unknown')}"] += 1 

241 if not buckets: 

242 return f"No IIP events in the last {weeks} week(s)." 

243 lines = [] 

244 for week in sorted(buckets): 

245 b = buckets[week] 

246 parts = [f"{key}={count}" for key, count in sorted(b.items())] 

247 lines.append(f"{week}: " + " ".join(parts)) 

248 return "\n".join(lines)
