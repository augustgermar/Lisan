1"""Resolving one record's reference to another. Deterministic, one home. 

2 

3`links`, `supporting_records`, and friends hold references between records. 

4Three things were quietly wrong with how they were produced and checked, and 

5they accounted for 1298 of a production vault's 1470 validation errors — 

6enough noise that the validator stopped being read at all, which is the real 

7cost. 

8 

91. **Archiving broke references that were never broken.** Epoch archiving and 

10 entity merges move a record under ``archive/`` and deliberately *preserve 

11 its id*. The validator's id index excluded archive — correctly, because 

12 its other job is auditing duplicate ids, and archived copies legitimately 

13 share ids with the live record. That exclusion leaked into link checking, 

14 so every reference to an archived record read as dangling. 695 errors, 

15 all of them pointing at records that exist. 

16 

172. **References came straight from the model.** The analyst's 

18 ``supporting_records`` was passed through from LLM output with no 

19 resolution, so a model that had seen ``episodes/2026-07-05-….md`` in its 

20 context would emit the bare filename, or invent 

21 ``entities/people/example-person.md`` for an entity actually filed as 

22 ``entity.example-person-with-surnames``. 561 errors. Deterministic-first says the 

23 model may *propose* a reference; code decides what it resolves to. 

24 

253. **Prose ended up in a reference list.** 42 entries were whole sentences 

26 ("Greenhouse irrigation was replaced on the second of the month."). Reported with the same 

27 message as a genuine dangling reference, which buried the difference 

28 between "this target moved" and "this field contains the wrong kind of 

29 thing entirely". 

30 

31:func:`resolve_reference` is the shared answer to "what does this string 

32point at?", used by the validator to judge and by the record factory to 

33canonicalize before writing. 

34"""

44# A reference is an id or a path. Anything with sentence punctuation, spaces 

45# beyond a couple, or sheer length is prose that leaked into the field — a 

46# different defect from a target that moved, and worth a different message. 

47_MAX_REFERENCE_LENGTH = 200

51# therefore into possible graph nodes. Mirrors the validator's structured-record 

52# rule. `archive/` belongs here: archived snapshots stay in files and FTS by 

53# design (rebuild_index.py: "Archived entity snapshots stay searchable"), which 

54# is exactly why a reference to an archived id still resolves.

56 "entities", "episodes", "knowledge", "evidence", "state", "decisions",

62def is_indexable(rel: Path) -> bool: 

63 """Can a record at this vault-relative path become a graph node? 

64 

65 `drafts/` cannot: the indexer skips them, so they hold no row in `files` 

66 and no reference to one will ever join. Rewriting a draft reference into a 

67 draft id would swap one non-joining string for another — churn dressed as 

68 a repair, which a first dry run of this migration proposed across 286 files 

69 before this check existed. 

70 """ 

71 return bool(rel.parts) and rel.parts[0] in INDEXED_DIRECTORIES

74@dataclass(slots=True) 

75class ReferenceIndex: 

76 """Everything a reference could legitimately name, built once per pass.""" 

77 

78 ids: dict[str, Path] = field(default_factory=dict) 

79 """Indexable record ids, live and archived — the set that can be a graph 

80 node. Archiving preserves ids by design, so archived targets belong here.""" 

81 

82 live_ids: dict[str, Path] = field(default_factory=dict) 

83 """Live records only. The duplicate-id audit needs this narrower view.""" 

84 

85 by_basename: dict[str, list[str]] = field(default_factory=dict) 

86 """Filename -> record ids, for repairing model-shortened paths. A name 

87 matching more than one record is ambiguous and never auto-resolved.""" 

88 

89 unindexed_paths: dict[str, str] = field(default_factory=dict) 

90 """Vault-relative path -> id, for records that exist but are not graph 

91 nodes (drafts). Kept so a reference to one gets an accurate message rather 

92 than "no such record".""" 

93 

94 merged_into: dict[str, str] = field(default_factory=dict) 

95 """Retired id -> surviving id, from an archived record's ``merged_into``. 

96 

97 A merge preserves the fragment's id in the archive, so a reference to it 

98 still resolves and nothing looks broken — while retrieval reaches a stub 

99 instead of the person the fragment became. Following the forwarding 

100 address is the difference between a reference that works and one that 

101 merely validates.""" 

102 

103 by_claim_text: dict[str, list[str]] = field(default_factory=dict) 

104 """Exact ``claim_text`` -> claim ids. 

105 

106 The writer put claim *text* where claim *ids* belong, in `links` and 

107 `linked_claims` alike, so 39 references were whole sentences. They are not 

108 junk: each is the verbatim text of a claim record that exists, which makes 

109 them repairable by exact match rather than by guesswork. Only exact, only 

110 unique — a near-match between two claims is a judgement, not a lookup.""" 

111 

112 by_entity_alias: dict[str, list[str]] = field(default_factory=dict) 

113 """Slugified entity name/alias -> entity ids. 

114 

115 The last resort for a reference the model invented rather than copied: 

116 ``entities/people/rosa.md`` names a real person filed under her full name. 

117 Aliases are the vault's own record of what she is called, so matching 

118 through them is reading the data rather than guessing at it — and the same 

119 ambiguity rule applies, because three people sharing a first name make it 

120 unresolvable, not a coin flip.""" 

121 

122 _path_to_id: dict[str, str] = field(default_factory=dict) 

123 

124 def id_for_path(self, rel: str) -> str | None: 

125 return self._path_to_id.get(rel)

127 def survivor_of(self, record_id: str) -> str | None: 

128 """Follow merge forwarding to the final survivor, cycle-safe.""" 

129 seen: set[str] = set() 

130 current = record_id 

131 while current in self.merged_into: 

132 if current in seen: 

133 return None 

134 seen.add(current) 

135 nxt = self.merged_into[current] 

136 if not nxt or nxt == current: 

137 return None 

138 current = nxt 

139 return current if current != record_id else None

142def build_reference_index(vault: Path) -> ReferenceIndex: 

143 index = ReferenceIndex() 

144 for path in iter_markdown_files(vault): 

145 try: 

146 rel = path.relative_to(vault) 

147 except ValueError: 

148 continue 

149 try: 

150 frontmatter = load_frontmatter_cached(path) 

151 except (FrontmatterError, OSError): 

152 continue 

153 record_id = frontmatter.get("id") 

154 if not isinstance(record_id, str) or not record_id.strip(): 

155 continue 

156 record_id = record_id.strip() 

157 rel_str = rel.as_posix() 

158 if not is_indexable(rel): 

159 index.unindexed_paths.setdefault(rel_str, record_id) 

160 continue 

161 # A live record outranks an archived one carrying the same id. Archived 

162 # copies legitimately share ids (epoch snapshots, merges, superseded 

163 # revisions), and `setdefault` alone would let whichever the directory 

164 # walk reached first win — so a reference could resolve to a retired 

165 # snapshot while the current record sat right there. 

166 archived = "archive" in rel.parts 

167 if record_id not in index.ids or (not archived and "archive" in index.ids[record_id].parts): 

168 index.ids[record_id] = rel 

169 index._path_to_id.setdefault(rel_str, record_id) 

170 forwarding = frontmatter.get("merged_into") 

171 if isinstance(forwarding, str) and forwarding.strip() and forwarding.strip() != record_id: 

172 index.merged_into.setdefault(record_id, forwarding.strip()) 

173 if "archive" not in rel.parts: 

174 index.live_ids.setdefault(record_id, rel) 

175 index.by_basename.setdefault(path.name, []).append(record_id) 

176 claim_text = frontmatter.get("claim_text") 

177 if isinstance(claim_text, str) and claim_text.strip(): 

178 key = " ".join(claim_text.split()) 

179 index.by_claim_text.setdefault(key, []).append(record_id) 

180 if str(frontmatter.get("type")) == "entity": 

181 names = [frontmatter.get("canonical_name"), frontmatter.get("nickname")] 

182 names.extend(frontmatter.get("aliases") or []) 

183 for name in names: 

184 key = _alias_key(name) 

185 if key and record_id not in index.by_entity_alias.setdefault(key, []): 

186 index.by_entity_alias[key].append(record_id) 

187 return index

190def _alias_key(value: Any) -> str: 

191 """Comparison form for a name: lowercased, non-alphanumerics to hyphens. 

192 

193 Makes ``Rosa Delgado``, ``rosa-delgado`` and ``Rosa_Delgado`` one key, 

194 which is what lets a reference written as a filename find the person. 

195 """ 

196 if not isinstance(value, str): 

197 return "" 

198 import re as _re 

199 

200 return _re.sub(r"[^a-z0-9]+", "-", value.strip().lower()).strip("-")

203def looks_like_prose(value: str) -> bool: 

204 """True when a reference slot holds a sentence rather than a reference. 

205 

206 Deliberately conservative: a real id or path has no spaces at all, so a 

207 single space is already suspicious, but ids are generated from slugified 

208 summaries and can be long, so length alone is not the test. 

209 """ 

210 text = value.strip() 

211 if not text: 

212 return False 

213 if len(text) > _MAX_REFERENCE_LENGTH: 

214 return True 

215 if text.count(" ") >= 3: 

216 return True 

217 # Sentence-enders and list bullets never appear in an id or a path. 

218 return text.endswith(".") and " " in text or text.startswith("- ")

221@dataclass(slots=True) 

222class Resolution: 

223 kind: str # "id" | "path" | "basename" | "prose" | "unknown" 

224 target: str = "" # the canonical id when resolvable 

225 detail: str = "" # for messages the owner reads 

226 

227 @property 

228 def ok(self) -> bool: 

229 """Traversable **as written**. 

230 

231 Only an exact id qualifies, and this is not pedantry. SPEC.md declares 

232 ``links(target_id) REFERENCES files(id)``, and the indexer inserts the 

233 raw reference string as ``target_id``, so graph retrieval joins it 

234 against ``files.id``. A path or filename therefore produces an edge 

235 that matches nothing: on the vault that prompted this work, 1349 of 

236 2860 edges were dead that way, and every one of them *passed* 

237 validation because the old check asked whether the file existed rather 

238 than whether the reference resolved. An instrument that answers a 

239 different question than the one that matters is worse than silence. 

240 """ 

241 return self.kind == "id" 

242 

243 @property 

244 def repairable(self) -> bool: 

245 """Names a real record, but not in a form the graph can follow. 

246 ``lisan migrate refs`` rewrites these to :attr:`target`.""" 

247 return self.kind in {"path", "basename", "merged", "alias", "claim_text"} and bool(self.target)

250def resolve_reference(value: Any, vault: Path, index: ReferenceIndex) -> Resolution: 

251 """What does *value* point at? 

252 

253 Resolution order — exact id, then vault-relative path, then a unique 

254 basename match. The path and basename steps do not bless the reference; 

255 they identify what it *meant*, so the migration can rewrite it to an id. 

256 A basename matching more than one record stays unresolved, because 

257 guessing between two records is how a wrong reference becomes a confident 

258 one. 

259 """ 

260 if not isinstance(value, str): 

261 return Resolution("unknown", detail=f"{type(value).__name__} is not a reference") 

262 text = value.strip() 

263 if not text: 

264 return Resolution("unknown", detail="empty reference") 

265 if text in index.ids: 

266 survivor = index.survivor_of(text) 

267 if survivor and survivor in index.ids: 

268 # Resolvable, but pointing at a fragment that was merged into 

269 # someone else. Repairable rather than fine: the reference works 

270 # and still reaches the wrong record. 

271 return Resolution( 

272 "merged", 

273 target=survivor, 

274 detail=f"{text} was merged into {survivor}", 

275 ) 

276 return Resolution("id", target=text) 

277 if text in index.unindexed_paths: 

278 # A real file, but not a graph node — a draft. Rewriting it to its id 

279 # would not make the edge join, so it is left exactly as written. 

280 return Resolution( 

281 "unindexed", 

282 detail="points at a draft, which the indexer does not make a graph node", 

283 ) 

284 if (vault / text).exists(): 

285 canonical = index.id_for_path(text) 

286 if canonical: 

287 return Resolution("path", target=canonical, detail=f"path form of {canonical}") 

288 # The file is really there, but it is not a record the indexer makes a 

289 # node of — a transcript, a primer document. Evidence citing the 

290 # transcript it was extracted from is provenance, and provenance to a 

291 # real file is not a broken reference. Calling it one turned ~150 

292 # correct citations into errors on first measurement. 

293 return Resolution( 

294 "unindexed", 

295 detail="points at a real file that is not an indexed record (provenance, not a graph edge)", 

296 ) 

297 if looks_like_prose(text): 

298 matches = index.by_claim_text.get(" ".join(text.split())) or [] 

299 if len(matches) == 1: 

300 return Resolution( 

301 "claim_text", 

302 target=matches[0], 

303 detail=f"verbatim claim_text of {matches[0]}", 

304 ) 

305 if len(matches) > 1: 

306 return Resolution( 

307 "prose", 

308 detail=f"text matches {len(matches)} claims; resolve by hand", 

309 ) 

310 return Resolution("prose", detail="prose in a reference field") 

311 basename = text.rsplit("/", 1)[-1] 

312 if basename.endswith(".md"): 

313 matches = index.by_basename.get(basename) or [] 

314 if len(matches) == 1: 

315 return Resolution("basename", target=matches[0], detail=f"filename of {matches[0]}") 

316 if len(matches) > 1: 

317 return Resolution("unknown", detail=f"filename {basename!r} matches {len(matches)} records; ambiguous") 

318 # Last resort: the reference may name an entity by a name it actually goes 

319 # by, in a file that never existed — a model writing 

320 # `entities/people/<firstname>.md` for someone filed under a full name. 

321 stem = basename[:-3] if basename.endswith(".md") else basename 

322 alias_matches = index.by_entity_alias.get(_alias_key(stem)) or [] 

323 if len(alias_matches) == 1: 

324 return Resolution( 

325 "alias", 

326 target=alias_matches[0], 

327 detail=f"names {alias_matches[0]} by one of its aliases", 

328 ) 

329 if len(alias_matches) > 1: 

330 return Resolution( 

331 "unknown", 

332 detail=f"{stem!r} is an alias of {len(alias_matches)} entities; ambiguous — resolve by hand", 

333 ) 

334 return Resolution("unknown", detail="no record with this id, path, or filename")

337def canonicalize_references( 

338 values: list[Any] | None, 

339 vault: Path, 

340 index: ReferenceIndex | None = None, 

341) -> tuple[list[str], list[str]]: 

342 """Rewrite a reference list to canonical ids. Returns (kept, dropped). 

343 

344 Used at write time so a model's approximate reference becomes an exact one 

345 before it lands, and unresolvable entries never land at all. The caller 

346 logs what was dropped — silently discarding a reference the model believed 

347 in is the same dishonesty as inventing one. 

348 """ 

349 if not values: 

350 return [], [] 

351 index = index or build_reference_index(vault) 

352 kept: list[str] = [] 

353 dropped: list[str] = [] 

354 for value in values: 

355 resolution = resolve_reference(value, vault, index) 

356 if resolution.ok or resolution.repairable: 

357 # Both land as the canonical id: an id stays itself, a path or 

358 # filename becomes the id it was reaching for. This is the step 

359 # that makes a model's approximate reference exact before it is 

360 # written, rather than after it has broken the graph. 

361 if resolution.target and resolution.target not in kept: 

362 kept.append(resolution.target) 

363 else: 

364 dropped.append(f"{value!r}: {resolution.detail}" if isinstance(value, str) else repr(value)) 

365 return kept, dropped
