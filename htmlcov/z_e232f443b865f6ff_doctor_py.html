1"""`lisan doctor`: check that the system's claims about itself are true. 

2 

3The project's principle is "instruments before rules": an honesty rule is 

4worthless without something that can tell the truth. The doctor applies 

5that to the system's own architecture. Each check states a claim the README 

6or SPEC makes and measures whether the running install keeps it: 

7 

8- every outbound or execution path is decided by the authority engine; 

9- every declared job type is dispatchable; 

10- the hand-written "not built" list does not deny a real capability; 

11- intent.md is adopted, so execution is not silently blocked or open; 

12- the identity kernel matches its ratified hash; 

13- entity relationships survive a rebuild (they live in markdown); 

14- every record marked "embedded" has a vector; 

15- entity story logs hold the owner's words, not pipeline JSON; 

16- self-repair cannot touch the mechanisms that judge it. 

17 

18Read-only: the doctor never repairs anything. It reports PASS, WARN, or 

19FAIL with the evidence, so the owner (or a self-repair loop) can act. 

20"""

43def _check_authority_covers_outbound(vault: Path, db_path: Path) -> Check: 

44 from .authority import ACTIONS 

45 from .execution_tools import TOOLS 

46 

47 claim = "Every action that leaves the machine needs the owner's yes for that exact action." 

48 outbound = {name: spec for name, spec in ACTIONS.items() if spec.outbound} 

49 unapproved = [name for name, spec in outbound.items() if not spec.approval and not name.startswith("drive.")] 

50 tool_names = {t["name"] for t in TOOLS} 

51 missing = [name for name in ("send_email",) if name in tool_names and name not in ACTIONS] 

52 if unapproved or missing: 

53 return Check("authority", claim, FAIL, f"ungated outbound actions: {unapproved + missing}") 

54 return Check("authority", claim, PASS, f"{len(outbound)} outbound action kinds, all approval-gated or tier-gated")

57def _check_job_registry(vault: Path, db_path: Path) -> Check: 

58 from .jobs import _JOB_HANDLERS, JOB_TYPES 

59 

60 claim = "Every declared job type can be dispatched." 

61 missing = sorted(set(JOB_TYPES) - set(_JOB_HANDLERS)) 

62 extra = sorted(set(_JOB_HANDLERS) - set(JOB_TYPES)) 

63 if missing or extra: 

64 return Check("jobs", claim, FAIL, f"undispatchable: {missing}; undeclared handlers: {extra}") 

65 return Check("jobs", claim, PASS, f"{len(JOB_TYPES)} job types, one handler each")

68def _check_not_built(vault: Path, db_path: Path) -> Check: 

69 from .execution_tools import TOOLS 

70 from .self_model import NOT_BUILT 

71 

72 claim = "The self-model never tells the agent a real capability is missing." 

73 tool_names = {t["name"].replace("_", " ") for t in TOOLS} 

74 text = " ".join(f"{i['name']} {i['detail']}" for i in NOT_BUILT).lower() 

75 contradicted = sorted(name for name in tool_names if len(name) > 6 and f"no {name}" in text) 

76 if contradicted: 

77 return Check("self-model", claim, FAIL, f"NOT_BUILT denies: {contradicted}") 

78 return Check("self-model", claim, PASS, f"{len(NOT_BUILT)} declared gaps, none contradicted by the tool surface")

81def _check_intent(vault: Path, db_path: Path) -> Check: 

82 from .intent import IntentError, has_sentinel_dates, load_intent 

83 

84 claim = "Execution answers to an adopted intent.md (absent or template intent blocks execution)." 

85 try: 

86 intent = load_intent(vault) 

87 except IntentError as exc: 

88 return Check("intent", claim, WARN, f"{exc} — chat execution and delegation will refuse") 

89 if has_sentinel_dates(intent): 

90 return Check("intent", claim, WARN, "intent.md is still the template — execution will refuse until adopted") 

91 return Check("intent", claim, PASS, f"intent.md v{intent.version} adopted")

94def _check_kernel(vault: Path, db_path: Path) -> Check: 

95 from .kernel import compute_kernel_hash, kernel_path, stored_kernel_hash 

96 

97 claim = "The identity kernel changes only through a ceremony." 

98 path = kernel_path(vault) 

99 if not path.exists(): 

100 return Check("kernel", claim, WARN, "no identity kernel yet") 

101 # Read-only: verify_kernel would also append a drift report. 

102 text = path.read_text(encoding="utf-8") 

103 stored = stored_kernel_hash(text) 

104 if not stored: 

105 return Check("kernel", claim, WARN, "kernel is unstamped") 

106 if compute_kernel_hash(text) != stored: 

107 return Check("kernel", claim, FAIL, "kernel content no longer matches its ratified hash (drift)") 

108 return Check("kernel", claim, PASS, "kernel matches its ratified hash")

119def _check_edges_in_markdown(vault: Path, db_path: Path) -> Check: 

120 from ..frontmatter import load_frontmatter_cached 

121 

122 claim = "Entity relationships live in markdown, so a rebuild restores them." 

123 conn = _connect(db_path) 

124 if conn is None: 

125 return Check("edges", claim, WARN, "no index database") 

126 try: 

127 rows = conn.execute( 

128 "SELECT l.source_id, l.target_id, l.relationship_type, f.path FROM links l " 

129 "JOIN files f ON f.id = l.source_id WHERE l.source_id LIKE 'entity.%' AND l.target_id LIKE 'entity.%'" 

130 ).fetchall() 

131 except sqlite3.Error as exc: 

132 return Check("edges", claim, WARN, f"could not read links: {exc}") 

133 finally: 

134 conn.close() 

135 orphaned = 0 

136 for source_id, target_id, rel_type, rel_path in rows: 

137 try: 

138 rels = load_frontmatter_cached(vault / rel_path).get("relationships") or [] 

139 except Exception: 

140 rels = [] 

141 if not any(isinstance(r, dict) and r.get("target") == target_id for r in rels): 

142 orphaned += 1 

143 if orphaned: 

144 return Check("edges", claim, FAIL, f"{orphaned} of {len(rows)} entity edges exist only in SQLite") 

145 return Check("edges", claim, PASS, f"{len(rows)} entity edges, all backed by frontmatter")

148def _check_embeddings(vault: Path, db_path: Path) -> Check: 

149 from .vector_store import load_index 

150 

151 claim = "A record marked 'embedded' has a vector on disk." 

152 conn = _connect(db_path) 

153 if conn is None: 

154 return Check("embeddings", claim, WARN, "no index database") 

155 try: 

156 embedded = {r[0] for r in conn.execute("SELECT id FROM files WHERE embedding_status = 'embedded'")} 

157 except sqlite3.Error as exc: 

158 return Check("embeddings", claim, WARN, f"could not read files: {exc}") 

159 finally: 

160 conn.close() 

161 # Records may be stored whole (``<id>``) or as passages 

162 # (``<id>::passage:<n>:<hash>``); either form means the record has a vector. 

163 vectors = {key.split("::passage:", 1)[0] for key in load_index(db_path.parent / "embeddings.bin").vectors} 

164 missing = embedded - vectors 

165 if missing: 

166 status = FAIL if len(missing) > max(5, len(embedded) // 50) else WARN 

167 return Check("embeddings", claim, status, f"{len(missing)} of {len(embedded)} 'embedded' rows have no vector") 

168 return Check("embeddings", claim, PASS, f"{len(embedded)} embedded rows, all with vectors")

174def _check_story_logs(vault: Path, db_path: Path) -> Check: 

175 from ..frontmatter import load_frontmatter_cached 

176 

177 claim = "Entity story logs hold what was said, not pipeline JSON." 

178 total = polluted = 0 

179 affected = 0 

180 for path in (vault / "entities").rglob("*.md") if (vault / "entities").exists() else []: 

181 if "archive" in path.parts: 

182 continue 

183 try: 

184 log = load_frontmatter_cached(path).get("source_log") or [] 

185 except Exception: 

186 continue 

187 bad = sum(1 for e in log if isinstance(e, dict) and any(m in str(e.get("text", "")) for m in _PIPELINE_MARKERS)) 

188 total += len(log) 

189 polluted += bad 

190 affected += bool(bad) 

191 if polluted: 

192 return Check("story-logs", claim, WARN, 

193 f"{polluted} of {total} log entries ({affected} entities) carry pipeline JSON from before the " 

194 "2026-10-29 fix; new entries are clean") 

195 return Check("story-logs", claim, PASS, f"{total} log entries, none with pipeline JSON")

198def _check_self_repair_protection(vault: Path, db_path: Path) -> Check: 

199 from .self_repair import SelfRepairRefused, validate_patch_paths 

200 

201 claim = "Self-repair cannot modify what decides, verifies, or undoes a repair." 

202 must_refuse = [ 

203 "lisan/tools/authority.py", "lisan/tools/intent.py", "lisan/tools/judge.py", 

204 "lisan/tools/kernel.py", "lisan/tools/purge.py", "tests/test_authority.py", 

205 ] 

206 allowed = [] 

207 for path in must_refuse: 

208 diff = f"diff --git a/{path} b/{path}\n--- a/{path}\n+++ b/{path}\n@@ -1 +1 @@\n-x\n+y\n" 

209 try: 

210 validate_patch_paths(diff) 

211 allowed.append(path) 

212 except SelfRepairRefused: 

213 pass 

214 if allowed: 

215 return Check("self-repair", claim, FAIL, f"patchable: {allowed}") 

216 return Check("self-repair", claim, PASS, f"{len(must_refuse)} sampled guard paths refused")

232def run_doctor(vault: Path, db_path: Path) -> list[Check]: 

233 results: list[Check] = [] 

234 for check in CHECKS: 

235 try: 

236 results.append(check(vault, db_path)) 

237 except Exception as exc: # a broken instrument is itself a finding 

238 name = check.__name__.removeprefix("_check_").replace("_", "-") 

239 results.append(Check(name, "(check crashed)", FAIL, f"{type(exc).__name__}: {exc}")) 

240 return results

243def format_doctor(results: list[Check]) -> str: 

244 width = max(len(r.name) for r in results) if results else 0 

245 lines = [] 

246 for r in results: 

247 lines.append(f"{r.status:4} {r.name:<{width}} {r.claim}") 

248 lines.append(f" {'':<{width}} → {r.evidence}") 

249 worst = FAIL if any(r.status == FAIL for r in results) else WARN if any(r.status == WARN for r in results) else PASS 

250 lines.append("") 

251 lines.append(f"overall: {worst}") 

252 return "\n".join(lines)
