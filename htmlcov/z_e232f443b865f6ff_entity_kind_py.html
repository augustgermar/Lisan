1"""Entity kind model (P3). 

2 

3`kind` is a first-class, OPEN property of every entity. Assignment is 

4deterministic-first, mirroring the deixis/routing spine: 

5 

6 roster (authoritative) -> structural signals (mechanical) -> model -> thing 

7 

8`person` is NEVER a default at any layer — forced classification with no escape 

9hatch is what made "Atlas" (a project) and "Houston" (a city) become *people*. 

10`thing` is the honest fallback so a wrong guess degrades to vagueness, not a 

11confident lie. The kind set is open: unknown kinds are accepted, not rejected. 

12""" 

13from __future__ import annotations 

14 

15import re 

16from pathlib import Path 

17 

18# Canonical starter set (§3). OPEN — new kinds may appear in roster/config and 

19# must not break the system. This is the validation hint, not a closed enum. 

20CANONICAL_KINDS: frozenset[str] = frozenset({ 

21 # animate / agentive 

22 "person", "pet", "agent", "organization", 

23 # concrete / physical 

24 "place", "system", "artifact", 

25 # abstract / conceptual 

26 "project", "event", "topic", "account", 

27 # fallback 

28 "thing", 

29}) 

30 

31FALLBACK_KIND = "thing" 

32 

33# ── Layer 2: structural signals (high-precision, deterministic) ─────────────── 

34 

35_IP = re.compile(r"^\d{1,3}(?:\.\d{1,3}){3}(?:/\d{1,2})?$") 

36_HOSTNAME = re.compile(r"^[a-z0-9]([a-z0-9-]*[a-z0-9])?(\.[a-z0-9]([a-z0-9-]*[a-z0-9])?)+$") 

37_URL = re.compile(r"^(https?|ssh|git)://", re.I) 

38_ORG_SUFFIX = re.compile( 

39 r"\b(inc|inc\.|llc|l\.l\.c\.|corp|corp\.|ltd|ltd\.|co\.|gmbh|plc|" 

40 r"university|college|institute|foundation|agency|department|dept\.?|" 

41 r"bureau|court|guild|union|company)\b", 

42 re.I, 

43) 

44_ACCOUNT = re.compile(r"^(?:\$\d[\d,]*(?:\.\d+)?|\d{4}[- ]?\d{4}[- ]?\d{4}[- ]?\d{4}|[A-Z]{2}\d{2}[A-Z0-9]{10,})$") 

45_FILE_EXT = re.compile(r"\.[A-Za-z0-9]{1,8}$") 

46_EVENT_WORDS = ( 

47 "birthday", "party", "recital", "wedding", "funeral", "graduation", 

48 "appointment", "deadline", "deploy", "deployment", "outage", "incident", 

49 "practice", "rehearsal", "ceremony", "anniversary", "hearing", "trial date", 

50) 

51

53def classify_structural(token: str, context: str = "") -> str | None: 

54 """Mechanically classify a token by shape. Returns a kind, or None if no 

55 high-precision pattern fires (defer to the model). Never guesses `person`.""" 

56 t = (token or "").strip() 

57 if not t: 

58 return None 

59 low = t.lower() 

60 if _IP.match(t): 

61 return "system" 

62 if _URL.match(t): 

63 return "system" 

64 if "/" in t or "\\" in t: 

65 # a path/repo ref: a file (has an extension) is an artifact; a dir/host 

66 # path is infrastructure. 

67 return "artifact" if _FILE_EXT.search(t) else "system" 

68 if _HOSTNAME.match(low) and (any(c.isdigit() for c in low) or "-" in low or low.rsplit(".", 1)[-1] in {"prod", "dev", "local", "test", "stage", "io", "com", "net", "org", "dev"}): 

69 return "system" 

70 if _ACCOUNT.match(t): 

71 return "account" 

72 if _ORG_SUFFIX.search(t): 

73 return "organization" 

74 # Kind describes what the entity IS, never what the turn was about: the 

75 # event check applies to the name alone. Scanning the surrounding context 

76 # here turned every person mentioned near the word "birthday" into an 

77 # event-kinded entity. 

78 if any(w in low for w in _EVENT_WORDS): 

79 return "event" 

80 return None

83# ── Three-layer assignment ──────────────────────────────────────────────────── 

84 

85def assign_kind( 

86 name: str, 

87 vault: Path, 

88 *, 

89 model_kind: str | None = None, 

90 summary: str = "", 

91 source_text: str = "", 

92) -> str: 

93 """Resolve an entity's kind: roster -> structural -> model -> thing. 

94 

95 `person` is only ever produced by the roster or by an explicit model choice, 

96 never as a fallback. An unrecognized model kind is accepted (open set) only 

97 if it is non-empty and not a bare default; otherwise we fall to `thing`. 

98 """ 

99 name = (name or "").strip() 

100 if not name: 

101 return FALLBACK_KIND 

102 

103 # Layer 1 — roster (authoritative) 

104 from .primer_index import roster_kind 

105 rk = roster_kind(vault, name) 

106 if rk: 

107 return rk 

108 

109 # Layer 2 — structural signals 

110 sk = classify_structural(name, context=f"{summary} {source_text}") 

111 if sk: 

112 return sk 

113 

114 # Layer 3 — the model's explicit choice (allowed to be person, or anything 

115 # open). A blank/missing kind is NOT trusted — it falls through to `thing`. 

116 mk = (model_kind or "").strip().lower() 

117 if mk and mk != "unknown": 

118 return mk 

119 

120 # Honest fallback — never `person`. 

121 return FALLBACK_KIND
