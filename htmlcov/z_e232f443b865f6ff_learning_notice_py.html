1"""Tell the owner, in a line or two, when the agent learns something on its own. 

2 

3Learning that the owner cannot see is learning they cannot trust or correct. 

4Four things count: a skill the loop wrote or changed, a skill it promoted or 

5flagged, a belief the dreamer revised on evidence, and a new ache (a deviation 

6the scanner raised about the agent itself). Each message is deliberately short; 

7the full record is in the learning artifacts and `lisan skills history`. 

8 

9Best effort by construction: a failed notification must never undo or block the 

10learning it reports. Only the resident vault can page the owner (the check lives 

11in escalation._notify_owner, which also honours LISAN_NO_OUTBOUND), so tests and 

12scratch vaults stay silent. Turn it off with ``learning.notify: false``. 

13""" 

14from __future__ import annotations 

15 

16from pathlib import Path 

17from typing import Any 

18 

19MAX_LINE = 150 

20MAX_LINES = 4 

21 

22 

23def enabled(config: dict[str, Any] | None) -> bool: 

24 return (config or {}).get("learning", {}).get("notify", True) is not False 

25 

26 

27def _line(text: Any) -> str: 

28 one = " ".join(str(text or "").split()) 

29 return one if len(one) <= MAX_LINE else one[: MAX_LINE - 1].rstrip() + "…" 

30 

31 

32def send(vault: Path, title: str, lines: list[str], *, config: dict[str, Any] | None = None) -> bool: 

33 """One message: a title and up to MAX_LINES bullets (the rest counted).""" 

34 lines = [_line(x) for x in lines if x] 

35 if not lines: 

36 return False 

37 try: 

38 if config is None: 

39 from ..config import load_config 

40 

41 config = load_config() 

42 if not enabled(config): 

43 return False 

44 shown = [f"• {x}" for x in lines[:MAX_LINES]] 

45 if len(lines) > MAX_LINES: 

46 shown.append(f"…and {len(lines) - MAX_LINES} more") 

47 from .escalation import _notify_owner 

48 

49 return bool(_notify_owner(f"{title}\n" + "\n".join(shown), chat_id=None, vault=vault)) 

50 except Exception as exc: # reporting must never break learning 

51 try: 

52 from .log import log_error 

53 

54 log_error(vault, "learning_notice", exc) 

55 except Exception: 

56 pass 

57 return False

60def skills_learned(vault: Path, review_id: str, verdicts: list[Any], lifecycle: list[dict[str, Any]], *, config=None) -> bool: 

61 lines: list[str] = [] 

62 for v in verdicts: 

63 applied = getattr(v, "applied", None) 

64 if applied is None: 

65 continue 

66 verb = {"create": "new skill", "patch": "updated skill", "edit": "updated skill"}.get(applied.op, f"{applied.op} skill") 

67 why = getattr(getattr(v, "op", None), "rationale", "") or "" 

68 lines.append(f"{verb} {applied.skill}: {why}" if why else f"{verb} {applied.skill}") 

69 for change in lifecycle or []: 

70 verb = "now trusts" if change.get("to") == "established" else "flagged" 

71 lines.append(f"{verb} its skill {change.get('skill')}: {change.get('reason', '')}") 

72 return send(vault, "🧠 I learned something", lines, config=config)

75def beliefs_revised(vault: Path, statements: list[str], *, config=None) -> bool: 

76 return send(vault, "🪞 I revised what I believe about myself", statements, config=config)

79def aches(vault: Path, found: list[tuple[str, str]], *, config=None) -> bool: 

80 return send(vault, "🩺 I noticed something off", [f"{klass}: {summary}" for klass, summary in found], config=config)

83def aches_healed(vault: Path, healed: list[tuple[str, str]], *, config=None) -> bool: 

84 """An ache that stopped being true. The scanner can only say that the 

85 condition is no longer detected, not what fixed it, and says exactly that.""" 

86 return send(vault, "💚 An ache cleared up (no longer detected)", [f"{klass}: {summary}" for klass, summary in healed], config=config)
