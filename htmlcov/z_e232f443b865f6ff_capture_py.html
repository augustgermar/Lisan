20def capture_text( 

21 vault: Path, 

22 text: str, 

23 conversation_id: str | None = None, 

24 speaker: str = "USER", 

25 provider: str | None = None, 

26 model: str | None = None, 

27 conversation_policy: dict[str, Any] | None = None, 

28 queue_background: bool = True, 

29 db_path: Path | None = None, 

30 append_response_to_transcript: bool = False, 

31 drain_jobs: bool | None = None, 

32) -> dict[str, Any]: 

33 trace = None 

34 trace_token = None 

35 created_trace = False 

36 if get_current_turn_trace() is None: 

37 turn_id = f"capture.{time.strftime('%Y%m%d%H%M%S')}.{uuid.uuid4().hex[:8]}" 

38 trace, trace_token = start_turn_trace(turn_id, text, "capture", False) 

39 created_trace = True 

40 # Drain pending entity story rewrites before pipeline so retrieval sees 

41 # fresh entity narratives. Non-fatal: a failed drain never blocks capture. 

42 # TODO: consider capping drain depth if multi-entity turns consistently 

43 # exceed latency budget. 

44 _drain_entity_rewrite_jobs(vault=vault, db_path=db_path, provider=provider, model=model) 

45 try: 

46 result = run_memory_pipeline( 

47 vault=vault, 

48 text=text, 

49 conversation_id=conversation_id, 

50 speaker=speaker, 

51 provider=provider, 

52 model=model, 

53 conversation_policy=conversation_policy, 

54 db_path=db_path, 

55 ) 

56 except Exception as exc: 

57 log_error(vault, "capture_text", exc) 

58 raise 

59 try: 

60 out = { 

61 "transcript_path": str(result.transcript_path), 

62 "draft_path": str(result.draft_path or ""), 

63 "mode": result.mode, 

64 "action": result.action, 

65 "listener": result.listener, 

66 "writer": result.writer or {}, 

67 "skeptic": result.skeptic or {}, 

68 "interlocutor": result.interlocutor or {}, 

69 "elicitor": result.elicitor or {}, 

70 "narrative_state_path": str(result.narrative_state_path or ""), 

71 "narrative_state": result.narrative_state or {}, 

72 "conversation_policy": conversation_policy or {}, 

73 } 

74 response_text = _extract_capture_response(result) 

75 if append_response_to_transcript and response_text: 

76 from .transcripts import append_transcript 

77 

78 append_transcript(vault=vault, conversation_id=conversation_id, speaker="LISAN", text=response_text) 

79 out["response"] = response_text 

80 if queue_background and result.action != "skip": 

81 from .jobs import enqueue_job 

82 

83 queued_jobs = enqueue_post_turn_jobs( 

84 result, vault=vault, text=text, conversation_id=conversation_id, db_path=db_path, 

85 ) 

86 # Enqueue story-rewrite jobs for entities that received new material. 

87 for entity_path in (result.entities_touched or []): 

88 try: 

89 from ..frontmatter import load_markdown as _load_md 

90 entity_fm = _load_md(entity_path).frontmatter 

91 entity_id = str(entity_fm.get("id") or "").strip() 

92 except Exception: 

93 entity_id = "" 

94 rewrite_payload: dict[str, Any] = { 

95 "vault": str(vault), 

96 "entity_path": str(entity_path), 

97 "entity_id": entity_id or str(entity_path), 

98 "draft_path": str(result.draft_path or ""), 

99 "transcript_path": str(result.transcript_path), 

100 "conversation_id": conversation_id, 

101 } 

102 rewrite_payload = {k: v for k, v in rewrite_payload.items() if v not in (None, "")} 

103 try: 

104 from .job_policy import priority_for_job_type as _prio 

105 job_id = enqueue_job( 

106 "entity.rewrite_story", 

107 rewrite_payload, 

108 priority=_prio("entity.rewrite_story"), 

109 db_path=db_path, 

110 ) 

111 queued_jobs.append({"job_type": "entity.rewrite_story", "job_id": job_id}) 

112 except Exception: 

113 pass 

114 record_jobs_queued(len(queued_jobs)) 

115 out["queued_jobs"] = queued_jobs 

116 # Drain the indexing/embedding jobs we just enqueued, in 

117 # process, before returning — so the records this turn wrote are 

118 # actually embedded and semantic retrieval works without a manual 

119 # `lisan jobs run`. The user-facing response (out["response"]) is 

120 # already finalized above, so the drain never delays response 

121 # composition; it only extends the call's total wall-time. Strictly 

122 # bounded to index jobs (no LLM-heavy analyst/dreamer), and strictly 

123 # non-fatal: a failed embed leaves its job queued for the next drain 

124 # and never propagates out of capture. 

125 out["drained_jobs"] = _drain_index_jobs( 

126 vault=vault, db_path=db_path, provider=provider, model=model, 

127 queued_jobs=queued_jobs, drain_jobs=drain_jobs, 

128 ) 

129 if created_trace and trace is not None: 

130 finalized = finalize_turn_trace(trace, db_path=db_path or sqlite_path(), vault=vault) 

131 out["trace_summary"] = finalized.summary() 

132 out["trace"] = finalized.as_dict() 

133 log_capture(vault, text, out) 

134 return out 

135 finally: 

136 if created_trace: 

137 reset_current_turn_trace(trace_token)

140def enqueue_post_turn_jobs( 

141 result, 

142 *, 

143 vault: Path, 

144 text: str, 

145 conversation_id: str | None, 

146 db_path: Path | None = None, 

147) -> list[dict[str, Any]]: 

148 """Plan and enqueue the post-turn background jobs for a captured turn. 

149 

150 This seam is where every maintenance drive hangs — analyst scans, 

151 dreamer compaction, the weekly self-eval, the daily deviation scan. 

152 Both capture paths must pass through it: when the observer path 

153 (capture.observe) bypassed it, none of those drives ran even once 

154 between 2026-07-05 and 2026-07-15 and nothing said so. ``result`` is 

155 the MemoryPipelineResult of the turn just captured. 

156 """ 

157 from .job_policy import which_jobs_for_turn 

158 from .jobs import enqueue_job 

159 

160 queued_jobs: list[dict[str, Any]] = [] 

161 turn_metadata = { 

162 "vault": str(vault), 

163 "db_path": str(db_path) if db_path else None, 

164 "conversation_id": conversation_id, 

165 "text": text, 

166 "action": result.action, 

167 "mode": result.mode, 

168 "reason": "memory capture wrote or updated records", 

169 "listener": result.listener, 

170 "writer": result.writer or {}, 

171 "draft_path": str(result.draft_path or ""), 

172 "transcript_path": str(result.transcript_path), 

173 "records_written": _count_created_records(result.writer or {}), 

174 "high_salience": _is_high_salience(result.listener, result.writer or {}), 

175 "self_analysis_requested": _is_self_analysis_requested(text), 

176 "explicit_memory_request": _is_memory_request(text), 

177 } 

178 job_specs = which_jobs_for_turn(turn_metadata, db_path=db_path) 

179 for job_spec in job_specs: 

180 if isinstance(job_spec, dict): 

181 job_type = str(job_spec.get("job_type") or "") 

182 payload = job_spec.get("payload") or {} 

183 priority = int(job_spec.get("priority") or 100) 

184 else: 

185 job_type, payload, priority = job_spec 

186 try: 

187 job_id = enqueue_job(job_type, payload, priority=priority, db_path=db_path) 

188 queued_jobs.append({"job_type": job_type, "job_id": job_id}) 

189 except Exception: 

190 continue 

191 return queued_jobs

194def _drain_entity_rewrite_jobs( 

195 *, 

196 vault: Path, 

197 db_path: Path | None, 

198 provider: str | None, 

199 model: str | None, 

200) -> dict[str, Any]: 

201 """Drain pending entity story rewrite jobs before the pipeline runs. 

202 

203 Non-fatal: any error is silently swallowed. Returns quickly if there are 

204 no pending jobs (just a cheap DB count query). This ensures entity stories 

205 are current before retrieval, without blocking when there's nothing to do. 

206 """ 

207 from .jobs import run_jobs_worker 

208 

209 try: 

210 result = run_jobs_worker( 

211 vault=vault, 

212 db_path=db_path, 

213 provider=provider, 

214 model=model, 

215 job_types={"entity.rewrite_story"}, 

216 ) 

217 count = result.get("processed_count", 0) 

218 return {"drained": bool(count), "processed_count": count} 

219 except Exception: 

220 return {"drained": False, "reason": "error"}

223def _drain_index_jobs( 

224 *, 

225 vault: Path, 

226 db_path: Path | None, 

227 provider: str | None, 

228 model: str | None, 

229 queued_jobs: list[dict[str, Any]], 

230 drain_jobs: bool | None, 

231) -> dict[str, Any]: 

232 """Drain the indexing/embedding jobs enqueued for this turn, in-process. 

233 

234 Resolves the drain_on_capture config knob when ``drain_jobs`` is None. 

235 Bounded (only INDEX_JOB_TYPES, never the LLM-heavy maintenance jobs) and 

236 non-fatal (any error is logged and swallowed; the failed job stays queued 

237 for the next drain). Returns a small summary dict for observability. 

238 """ 

239 from .jobs import INDEX_JOB_TYPES, run_jobs_worker 

240 

241 if drain_jobs is None: 

242 from ..config import load_config 

243 drain_jobs = bool(load_config().get("jobs", {}).get("drain_on_capture", True)) 

244 if not drain_jobs: 

245 return {"drained": False, "reason": "disabled"} 

246 

247 # Only bother if this turn actually queued an index job. 

248 if not any(str(j.get("job_type") or "") in INDEX_JOB_TYPES for j in queued_jobs): 

249 return {"drained": False, "reason": "no_index_jobs"} 

250 

251 try: 

252 result = run_jobs_worker( 

253 vault=vault, 

254 db_path=db_path, 

255 provider=provider, 

256 model=model, 

257 job_types=set(INDEX_JOB_TYPES), 

258 ) 

259 return { 

260 "drained": True, 

261 "processed_count": result.get("processed_count", 0), 

262 "success_count": result.get("success_count", 0), 

263 "failure_count": result.get("failure_count", 0), 

264 } 

265 except Exception as exc: 

266 # A failed embed must NEVER fail a capture (preserves P2). Log and move on. 

267 log_error(vault, "capture_text.drain_index_jobs", exc) 

268 return {"drained": False, "reason": f"error: {exc.__class__.__name__}"}

271def _extract_capture_response(result: Any) -> str: 

272 elicitor = getattr(result, "elicitor", None) or {} 

273 response_text = str(elicitor.get("response") or "").strip() 

274 if not response_text: 

275 interlocutor = getattr(result, "interlocutor", None) or {} 

276 response_text = str(interlocutor.get("response") or "").strip() 

277 return response_text

280def _count_created_records(writer: dict[str, Any]) -> int: 

281 count = 0 

282 for key in ("evidence_to_create", "claims_to_create", "claims", "open_loops_to_create", "decisions_to_create"): 

283 value = writer.get(key) 

284 if isinstance(value, list): 

285 count += len(value) 

286 return count

289def _is_high_salience(listener: dict[str, Any], writer: dict[str, Any]) -> bool: 

290 reasons = [str(reason).lower() for reason in (listener.get("reason") or []) if reason] 

291 if any("high-risk" in reason for reason in reasons): 

292 return True 

293 if str(writer.get("significance") or "").lower() == "high": 

294 return True 

295 return False

298def _is_self_analysis_requested(text: str) -> bool: 

299 lowered = text.lower() 

300 return any(phrase in lowered for phrase in ["self-analysis", "analyze myself", "analyze me", "what pattern", "recurring pattern"])

303def _is_memory_request(text: str) -> bool: 

304 lowered = text.strip().lower() 

305 return lowered.startswith("/remember") or lowered.startswith("/forget") or "remember this" in lowered or "fact correction" in lowered
