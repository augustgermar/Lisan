1"""entity_story — durable log + periodic compaction for entity narratives. 

2 

3The design, chosen after a scale test showed per-turn full-rewrite was both 

4lossy (it dropped the arc's ending under length pressure) and the throughput 

5bottleneck at volume: 

6 

7- Every mention APPENDS the new material to an append-only ``source_log`` in 

8 the entity's frontmatter. This is cheap (no LLM), deterministic, and the 

9 durable ground truth — entries are never deleted, only marked ``folded`` 

10 once compaction has woven them into the narrative. 

11- COMPACTION runs only when enough unfolded material has accumulated (or on 

12 demand): one writer call re-tells the narrative core from the prior story 

13 plus the unfolded log. Rare and batched instead of per-turn. 

14- Only source-log slices with a literal entity mention are added to that 

15 entity's search index. Unverified material remains durable in the log but 

16 cannot associate unrelated turn-wide text with the entity. 

17- Compaction is lossy only in the *rendering*, never in the *storage* — the 

18 narrative is a regenerable view over a log that keeps everything. 

19"""

31# Compact after this many unfolded log entries have accumulated. Overridable 

32# via config (memory.entity_compact_threshold). 

33_DEFAULT_COMPACT_THRESHOLD = 3 

34_LOG_ENTRY_MAX_CHARS = 1200 

35# Keep at most this many log entries in the entity file itself; older folded 

36# entries spill to an append-only archive. Without the spill, every read 

37# parses and every write rewrites an ever-growing frontmatter blob — the 

38# LSM-tree analogy this design was sold on truncates its log after 

39# compaction, and for a well-loved entity (a parent, a child) the blob 

40# otherwise grows for life. Overridable via config (memory.entity_log_keep). 

41_DEFAULT_LOG_KEEP = 40 

42_ATTRIBUTION_RESOLVER = "literal_entity_mention_v1"

65def source_log_archive_path(vault: Path, entity_path: Path) -> Path: 

66 """Where an entity's spilled log entries live: append-only JSONL, one 

67 entry per line, never rewritten. The durable half of the storage

72def _spill_folded_log(vault: Path, entity_path: Path, log: list[dict[str, Any]]) -> tuple[list[dict[str, Any]], int]: 

73 """Move the oldest FOLDED entries beyond the keep-window to the archive 

74 file. Only a folded prefix may spill — an unfolded entry has not been 

75 woven into the narrative yet and must stay in the file. Append is 

76 deduplicated by exact line, so a retried job never doubles the archive. 

77 Returns (remaining_log, spilled_count).""" 

78 import json as _json 

79 

80 keep = _log_keep() 

81 if len(log) <= keep: 

82 return log, 0 

83 spill: list[dict[str, Any]] = [] 

84 for entry in log[: len(log) - keep]: 

85 if not entry.get("folded"): 

86 break 

87 spill.append(entry) 

88 if not spill: 

89 return log, 0 

90 

91 path = source_log_archive_path(vault, entity_path) 

92 path.parent.mkdir(parents=True, exist_ok=True) 

93 existing: set[str] = set() 

94 if path.exists(): 

95 existing = {ln.strip() for ln in path.read_text(encoding="utf-8").splitlines() if ln.strip()} 

96 lines = [] 

97 for entry in spill: 

98 line = _json.dumps(entry, ensure_ascii=True, sort_keys=True) 

99 if line not in existing: 

100 lines.append(line) 

101 existing.add(line) 

102 if lines: 

103 with path.open("a", encoding="utf-8") as fh: 

104 fh.write("\n".join(lines) + "\n") 

105 return log[len(spill):], len(spill)

108def rewrite_entity_story( 

109 vault: Path, 

110 entity_path: Path, 

111 *, 

112 draft_path: Path | None = None, 

113 transcript_path: Path | None = None, 

114 conversation_id: str | None = None, 

115 provider: str | None = None, 

116 model: str | None = None, 

117 db_path: Path | None = None, 

118 force_compact: bool = False, 

119) -> dict[str, Any]: 

120 """Append this turn's material to the entity's durable log, and compact the 

121 narrative when enough has accumulated. Returns a summary dict for the job 

122 dispatcher.""" 

123 if not entity_path.exists(): 

124 return {"updated": False, "reason": "entity_path_missing", "entity_path": str(entity_path)} 

125 

126 doc = load_markdown(entity_path) 

127 fm = dict(doc.frontmatter) 

128 prior_body = doc.body 

129 

130 new_material = _read_draft_body(draft_path) or _read_transcript_tail(transcript_path) 

131 

132 log = [dict(e) for e in (fm.get("source_log") or []) if isinstance(e, dict)] 

133 appended = False 

134 if new_material: 

135 entry = _condense(new_material) 

136 if entry and (not log or log[-1].get("text") != entry): 

137 log.append({ 

138 "date": today_iso(), "text": entry, "folded": False, 

139 "attribution": _literal_attribution(new_material, fm), 

140 }) 

141 appended = True 

142 fm["source_log"] = log 

143 

144 unfolded = [e for e in log if not e.get("folded")] 

145 should_compact = force_compact or len(unfolded) >= _compact_threshold() 

146 

147 if not should_compact: 

148 if not appended: 

149 return {"updated": False, "reason": "no_new_material", "entity_path": str(entity_path)} 

150 # Persist the appended log and index it now — the new material is 

151 # searchable immediately, even before it is woven into the prose. 

152 write_markdown(entity_path, fm, prior_body) 

153 _reindex(entity_path, vault, db_path) 

154 return {"updated": True, "action": "appended", "unfolded": len(unfolded), 

155 "entity_path": str(entity_path)} 

156 

157 if not unfolded: 

158 return {"updated": False, "reason": "nothing_to_compact", "entity_path": str(entity_path)} 

159 

160 return _compact( 

161 vault=vault, entity_path=entity_path, fm=fm, prior_body=prior_body, 

162 log=log, unfolded=unfolded, provider=provider, model=model, db_path=db_path, 

163 )

166def _compact( 

167 *, 

168 vault: Path, 

169 entity_path: Path, 

170 fm: dict[str, Any], 

171 prior_body: str, 

172 log: list[dict[str, Any]], 

173 unfolded: list[dict[str, Any]], 

174 provider: str | None, 

175 model: str | None, 

176 db_path: Path | None, 

177) -> dict[str, Any]: 

178 """Re-tell the narrative core from the prior story plus the unfolded log, 

179 then mark those entries folded (never deleted). Guarded so a compaction can 

180 never return a materially shorter story than it replaced.""" 

181 from ..agents.writer import WriterAgent 

182 from ..schemas import get_schema 

183 

184 canonical_name = str(fm.get("canonical_name") or fm.get("id") or entity_path.stem) 

185 prior_story = _strip_title(prior_body).strip() 

186 new_material = "\n\n".join(f"({e.get('date')}) {e.get('text')}" for e in unfolded) 

187 

188 entity_meta = "\n".join([ 

189 f"canonical_name: {canonical_name}", 

190 f"kind: {fm.get('kind') or fm.get('subtype') or 'thing'}", 

191 f"summary: {fm.get('summary', '')}", 

192 f"significance: {fm.get('significance', 'low')}", 

193 ]) 

194 

195 agent = WriterAgent(vault=vault) 

196 result = agent.run_json( 

197 new_material, 

198 task="entity_story", 

199 schema=get_schema("entity_story_output"), 

200 provider=provider, 

201 model=model, 

202 entity_frontmatter=entity_meta, 

203 prior_story=prior_story or "(no prior story — first narrative write)", 

204 today=today_iso(), 

205 ) 

206 narrative = str(result.get("narrative") or "").strip() 

207 arc_note = str(result.get("arc_note") or "").strip() 

208 significance = str(result.get("significance") or "").strip().lower() 

209 if significance not in {"high", "medium", "low"}: 

210 significance = str(fm.get("significance") or "low").lower() 

211 significance_rationale = str(result.get("significance_rationale") or "").strip() 

212 updated_summary = str(result.get("summary") or "").strip() 

213 if not narrative: 

214 # An empty narrative is a provider failure (nine of them in one 

215 # burst on 2026-07-05), not a result — raise so the queue's retry 

216 # machinery owns it instead of recording a hollow success. 

217 raise RuntimeError(f"writer returned an empty narrative for {entity_path.name}") 

218 

219 # No-shrink guardrail: a compaction that folds in new material must never 

220 # come back materially shorter than the story it replaces. If it does, keep 

221 # the fuller prior story and append the new developments — the durable log 

222 # already holds everything, so this only protects the readable prose. 

223 prior_words = len(prior_story.split()) 

224 new_words = len(narrative.split()) 

225 if prior_words >= 60 and new_words < prior_words * 0.85: 

226 appended = _append_developments(prior_story, new_material) 

227 if appended: 

228 narrative = appended 

229 arc_note = (arc_note + " [guardrail: rewrite shrank; kept prior + appended new]").strip() 

230 else: 

231 return {"updated": False, "reason": "rewrite_shrank_story", 

232 "prior_words": prior_words, "new_words": new_words, 

233 "entity_path": str(entity_path)} 

234 

235 narrative = tokenize_principal(narrative, vault) 

236 # Mark the compacted entries folded — kept, not deleted — then spill 

237 # the oldest folded entries past the keep-window to the archive file, 

238 # so the in-file log stays bounded while the storage keeps everything. 

239 for entry in log: 

240 if not entry.get("folded"): 

241 entry["folded"] = True 

242 log, spilled = _spill_folded_log(vault, entity_path, log) 

243 fm["source_log"] = log 

244 fm["significance"] = significance 

245 if significance_rationale: 

246 fm["significance_rationale"] = significance_rationale 

247 if updated_summary: 

248 fm["summary"] = updated_summary 

249 fm["updated"] = today_iso() 

250 write_markdown(entity_path, fm, f"# {canonical_name}\n\n{narrative}\n") 

251 _reindex(entity_path, vault, db_path) 

252 return {"updated": True, "action": "compacted", "arc_note": arc_note, 

253 "folded": len(unfolded), "spilled": spilled, "entity_path": str(entity_path)}

256def entity_search_text(fm: dict[str, Any], body: str, archive_path: Path | None = None) -> str: 

257 """Narrative plus source-log slices justified by a literal entity mention. 

258 

259 Legacy entries are classified on read using canonical names and aliases; 

260 the Markdown source remains untouched. Unverified entries remain durable 

261 but do not inject unrelated turn-wide text into this entity's index row. 

262 """ 

263 import json as _json 

264 

265 parts = [_strip_title(body).strip()] 

266 for entry in (fm.get("source_log") or []): 

267 if isinstance(entry, dict): 

268 parts.extend(_attributed_slices(entry, fm)) 

269 if archive_path is not None and archive_path.exists(): 

270 try: 

271 for line in archive_path.read_text(encoding="utf-8").splitlines(): 

272 if not line.strip(): 

273 continue 

274 entry = _json.loads(line) 

275 if isinstance(entry, dict): 

276 parts.extend(_attributed_slices(entry, fm)) 

277 except Exception: 

278 pass 

279 return "\n\n".join(p for p in parts if p).strip()

282def shared_index_text(fm: dict[str, Any], body: str, archive_path: Path | None = None, *, limit: int = 64000) -> str: 

283 """Return the bounded, provenance-preserving text shared by lexical and 

284 semantic indexes. 

285 

286 Entity source logs are append-only and may outlive narrative compaction; 

287 using this one representation for both indexes prevents FTS-only facts 

288 from becoming invisible to paraphrased retrieval. The bound protects 

289 provider/index memory while retaining the newest tail and the narrative. 

290 """ 

291 text = entity_search_text(fm, body, archive_path=archive_path) 

292 if len(text) <= limit: 

293 return text 

294 narrative = _strip_title(body).strip() 

295 tail = text[-max(0, limit - len(narrative) - 80):] 

296 return (narrative + "\n\n[older source passages bounded; see canonical record]\n\n" + tail).strip()[:limit]

299def _identity_names(fm: dict[str, Any]) -> list[str]: 

300 names = [fm.get("canonical_name"), fm.get("name")] 

301 aliases = fm.get("aliases") or [] 

302 if isinstance(aliases, str): 

303 aliases = [aliases] 

304 names.extend(aliases if isinstance(aliases, list) else []) 

305 return sorted({str(name).strip() for name in names if str(name or "").strip()}, key=len, reverse=True)

308def _literal_attribution(source: str, fm: dict[str, Any]) -> dict[str, Any]: 

309 """Capture literal name matches with offsets and their containing sentence.""" 

310 mentions: list[dict[str, Any]] = [] 

311 for name in _identity_names(fm): 

312 for match in re.finditer(rf"(?<![\w]){re.escape(name)}(?![\w])", source, flags=re.IGNORECASE): 

313 start, end = match.span() 

314 left = max(source.rfind(mark, 0, start) for mark in (".", "!", "?", "\n")) + 1 

315 stops = [pos for mark in (".", "!", "?", "\n") if (pos := source.find(mark, end)) >= 0] 

316 right = min(stops) + 1 if stops else len(source) 

317 mentions.append({"match": match.group(0), "start": start, "end": end, 

318 "excerpt": re.sub(r"\s+", " ", source[left:right]).strip()}) 

319 # Keep overlapping canonical/alias matches deterministic and avoid duplicate slices. 

320 unique = {(m["start"], m["end"]): m for m in mentions} 

321 return {"mentions": sorted(unique.values(), key=lambda item: (item["start"], item["end"])), 

322 "resolver": _ATTRIBUTION_RESOLVER, 

323 "status": "verified" if unique else "unverified"}

326def _attributed_slices(entry: dict[str, Any], fm: dict[str, Any]) -> list[str]: 

327 text = str(entry.get("text") or "") 

328 attribution = entry.get("attribution") 

329 if isinstance(attribution, dict): 

330 if attribution.get("status") == "unverified": 

331 return [] 

332 mentions = attribution.get("mentions") or [] 

333 slices = [str(m.get("excerpt") or "").strip() for m in mentions if isinstance(m, dict)] 

334 if slices: 

335 return slices 

336 return [] 

337 # Backfill behavior for old active/archive entries: deterministic literal 

338 # matching only; never ask a model to infer identity retroactively. 

339 derived = _literal_attribution(text, fm) 

340 return [str(m["excerpt"]) for m in derived["mentions"] if m.get("excerpt")]

349def _condense(text: str) -> str: 

350 # Pipeline scaffolding (draft headers, transcript conversation markers, 

351 # code fences) is plumbing, not life — seen polluting live source_logs 

352 # ('# Memory Draft ## Status Skeptic approved…'). Strip before storing. 

353 text = _SCAFFOLDING.sub(" ", text) 

354 text = re.sub(r"\s+", " ", text).strip() 

355 return text[:_LOG_ENTRY_MAX_CHARS]

362def _append_developments(prior_story: str, new_material: str) -> str: 

363 body = _strip_title(prior_story).strip() 

364 if not body: 

365 return "" 

366 addition = re.sub(r"\s+", " ", new_material).strip()[:900] 

367 return f"{body}\n\n{addition}" if addition else body

379def _read_draft_body(draft_path: Path | None) -> str: 

380 """The material a story learns from is what was actually said. 

381 

382 Pipeline drafts render Status, Task, and the Listener/Writer/Skeptic/ 

383 Interlocutor JSON first and the owner's words last, under "## Source 

384 Text". This returned the whole body, and _condense keeps the first 1,200 

385 characters — so source_log entries held agent JSON and the real text was 

386 cut off. Return the Source Text section when there is one.""" 

387 if not draft_path: 

388 return "" 

389 try: 

390 path = Path(draft_path) 

391 if not path.exists(): 

392 return "" 

393 body = load_markdown(path).body 

394 except Exception: 

395 return "" 

396 match = _SOURCE_TEXT_HEADING.search(body) 

397 if match: 

398 return body[match.end():].strip() 

399 return body.strip()
